# Kustomize Local Validation Guide
# =================================

# Install kustomize on Mac
brew install kustomize

# Validate K8s manifests (dry-run, no deploy)
cd /path/to/hybrid-cloud-infrastructure
kustomize build kubernetes/dev/deployments

# This prints combined YAML output:
# - Validates syntax
# - Shows what Flux will deploy
# - Catches errors before pushing

# Validate specific folder
kustomize build kubernetes/dev/deployments/infrastructure/namespaces

# Save output to file for review
kustomize build kubernetes/dev/deployments > /tmp/k8s-output.yaml

# Common errors this catches:
# - YAML syntax errors
# - Missing kustomization.yaml
# - Missing resource files listed in kustomization.yaml
# - Invalid K8s resource structure
