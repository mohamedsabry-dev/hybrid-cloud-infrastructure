# Playbook to config the time of all nodes: 
# In ansible/dev/playbooks/common/ntp.yml
---
- name: Configure Chrony on FreeIPA
  hosts: freeipa
  become: yes
  tasks:
    - name: Install chrony
      ansible.builtin.package:
        name: chrony
        state: present

    - name: Enable and start chronyd
      ansible.builtin.service:
        name: chronyd
        state: started
        enabled: yes

    - name: Deploy IPA chrony config
      ansible.builtin.copy:
        src: "../../config_files/chrony_ipa.conf"
        dest: /etc/chrony.conf
        owner: root
        group: root
        mode: '0644'
        backup: yes
      notify: Restart Chrony

  handlers:
    - name: Restart Chrony
      ansible.builtin.service:
        name: chronyd
        state: restarted

##### Deploy Against LXC #####
## Create As config ansible/dev/config_files/chrony_lxc.conf

# Use public NTP servers from the pool.ntp.org project.
# The 'iburst' option allows for faster initial synchronization.

# 1. Your internal source (For consistency across the domain)
server freeipa.lab.local iburst prefer
# server 10.0.60.10 iburst # uncomment if dns down

# 2. Reliable secondary public sources (For redundancy)
pool 0.pool.ntp.org iburst
pool 1.pool.ntp.org iburst

# 3. Cloudflare
server time.cloudflare.com iburst

# Record the rate at which the system clock gains/losses time.
driftfile /var/lib/chrony/drift

# Allow the system clock to be stepped in the first three updates
# if its offset is larger than 1 second.
makestep 1.0 3

# Enable kernel synchronization of the real-time clock (RTC).
# rtcsync # LXC has no RTC (hardware clock)

# Specify the directory of additional configuration files.
confdir /etc/chrony.conf.d

# Enable logging of system clock gains/losses.
logdir /var/log/chrony

# Stop bad estimates from upsetting the machine clock.
maxupdateskew 100.0
