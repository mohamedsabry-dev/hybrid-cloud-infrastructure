#!/bin/bash
# ============================================================================
# ANSIBLE - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# PLAYBOOK EXECUTION & OPTIONS
# ============================================================================

ansible-playbook site.yml -i inventory/prod  # Specific inventory
ansible-playbook site.yml --limit webservers  # Target specific hosts/groups
ansible-playbook site.yml --limit 'host1,host2'  # Multiple hosts
ansible-playbook site.yml --tags "deploy,config"  # Run specific tags only
ansible-playbook site.yml --skip-tags "slow"  # Skip specific tags
ansible-playbook site.yml -e "env=prod version=1.2.3"  # Extra vars (highest precedence)
ansible-playbook site.yml -e "@vars/prod.yml"  # Extra vars from file
ansible-playbook site.yml --check --diff  # Dry-run with changes preview
ansible-playbook site.yml --start-at-task="Install nginx"  # Resume from specific task
ansible-playbook site.yml -f 10  # Parallel execution (10 hosts at once)
ansible-playbook site.yml -u deploy --become -K  # SSH user + sudo with password prompt

# ============================================================================
# VERBOSE & DEBUG LEVELS
# ============================================================================
ansible-playbook site.yml -v  # Show task results
ansible-playbook site.yml -vv  # Show task input parameters
ansible-playbook site.yml -vvv  # Show command execution (most common for TS)
ansible-playbook site.yml -vvvv  # SSH/connection-level debugging
ansible-playbook site.yml -vvv --check --diff  # Full troubleshooting mode

# ============================================================================
# AD-HOC COMMANDS
# ============================================================================
ansible all -m ping -i inventory.ini && ansible all -a -i inventory.ini "ping uptime # With specific inventory
ansible webservers -i inventory.ini -m service -a "name=nginx state=restarted" --become -K # Run command (shell is default module) if need sudo
ansible webservers -i inventory.ini -m shell -a "uptime"  # Explicit shell module
ansible webservers -i inventory.ini -m copy -a "src=/local/file dest=/remote/path mode=0644"  # Copy file
ansible all -i inventory.ini -m setup  # Gather facts
ansible all -i inventory.ini -m setup -a "filter=ansible_distribution*"  # Filter specific facts

# ============================================================================
# DOCUMENTATION & EXPLORATION
# ============================================================================
ansible-doc shell  # Full module documentation
ansible-doc -s copy  # Snippet mode - parameters only (QUICK REFERENCE)
ansible-doc -l | grep user  # List and search modules
ansible-config dump --only-changed  # Show non-default config
ansible-config list  # List all config options

# ============================================================================
# INVENTORY OPERATIONS
# ============================================================================
ansible all --list-hosts  # List all hosts
ansible webservers --list-hosts  # List hosts in group
ansible-inventory --graph  # Show inventory hierarchy
ansible-inventory --list  # JSON format
ansible-inventory --list --yaml  # YAML format
ansible-inventory --host server1  # Show specific host vars

# ============================================================================
# VALIDATION & SYNTAX
# ============================================================================
ansible-playbook site.yml --syntax-check  # Check YAML syntax
ansible-playbook site.yml --list-tasks  # Show tasks to be executed
ansible-playbook site.yml --list-tags  # Show available tags
ansible-playbook site.yml --list-hosts  # Show affected hosts
ansible-playbook site.yml --check  # Dry-run (no changes)
ansible-playbook site.yml --diff  # Show file differences
ansible-playbook site.yml --check --diff --list-tasks  # Full pre-flight validation

# ============================================================================
# FACTS & VARIABLES
# ============================================================================
ansible hostname -m setup  # All facts from host
ansible hostname -m setup -a "filter=ansible_eth*"  # Network facts
ansible hostname -m setup -a "filter=ansible_memory*"  # Memory facts
ansible hostname -m setup -a "filter=ansible_distribution*"  # OS facts

# ============================================================================
# VAULT (SECRETS)
# ============================================================================
ansible-vault create secrets.yml  # Create encrypted file
ansible-vault edit secrets.yml  # Edit encrypted file
ansible-vault encrypt vars/pass.yml  # Encrypt existing file
ansible-vault decrypt vars/pass.yml  # Decrypt file
ansible-vault view secrets.yml  # View without editing
ansible-vault rekey secrets.yml  # Change password
ansible-playbook site.yml --ask-vault-pass  # Run with vault password prompt
ansible-playbook site.yml --vault-password-file ~/.vault_pass  # Run with password file
ansible-vault encrypt_string 'secret123' --name 'db_password'  # Encrypt inline variable

# ============================================================================
# ROLES & GALAXY
# ============================================================================
ansible-galaxy role init my-role  # Create role structure
ansible-galaxy role install geerlingguy.nginx  # Install role from Galaxy
ansible-galaxy role install -r requirements.yml  # Install from requirements file
ansible-galaxy role list  # List installed roles
ansible-galaxy collection install community.general  # Install collection

# ============================================================================
# ENVIRONMENT VARIABLES (QUICK OVERRIDES)
# ============================================================================
# export ANSIBLE_INVENTORY=/path/to/inventory
# export ANSIBLE_CONFIG=/path/to/ansible.cfg
# export ANSIBLE_HOST_KEY_CHECKING=False  # Skip SSH key verification (DEV ONLY)
# export ANSIBLE_REMOTE_USER=deploy
# export ANSIBLE_LOG_PATH=/tmp/ansible.log

# ============================================================================
# COMMON PRODUCTION WORKFLOWS
# ============================================================================
# Safe deployment pattern:
ansible-playbook site.yml --check --diff --limit staging  # 1. Check staging
ansible-playbook site.yml --limit staging  # 2. Deploy staging 
ansible-playbook site.yml --check --diff --limit production  # 3. Check production
ansible-playbook site.yml --limit production  # 4. Deploy production


# ============================================================================
# SYNTAX REFERENCE
# ============================================================================
# ansible <pattern> -m <module> -a "<args>" [options]
# ansible-playbook <playbook> [options]
# ansible-doc [-s] <module>
# ansible-vault <action> <file>
# ansible-galaxy <type> <action> [options]
# ansible-inventory <action> [options]
