#!/bin/bash
# ============================================================================
# FREEIPA - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# AUTHENTICATION & SESSION
# ============================================================================
kinit admin  # Get Kerberos ticket (login)
kinit username@REALM.COM  # Specific user
klist  # Show current Kerberos tickets
kdestroy  # Destroy tickets (logout)
ipa ping  # Test IPA server connectivity

# ============================================================================
# USER MANAGEMENT
# ============================================================================
ipa user-find  # List all users
ipa user-find --login=john  # Search by login
ipa user-show john  # Show user details
ipa user-add john --first=John --last=Doe --email=john@example.com  # Add user
ipa user-mod john --shell=/bin/bash  # Modify user
ipa user-del john  # Delete user
ipa user-disable john  # Disable user account
ipa user-enable john  # Enable user account
ipa passwd john  # Reset user password (prompts)
ipa user-status john  # Show login status and failures

# ============================================================================
# GROUP MANAGEMENT
# ============================================================================
ipa group-find  # List all groups
ipa group-show developers  # Show group details
ipa group-add developers --desc="Developer Team"  # Add group
ipa group-add-member developers --users=john,jane  # Add users to group
ipa group-remove-member developers --users=john  # Remove user from group
ipa group-del developers  # Delete group

# ============================================================================
# HOST MANAGEMENT
# ============================================================================
ipa host-find  # List all hosts
ipa host-show server01.example.com  # Show host details
ipa host-add server01.example.com --ip-address=192.168.1.10  # Add host
ipa host-mod server01.example.com --location="DC1"  # Modify host
ipa host-del server01.example.com  # Delete host
ipa host-disable server01.example.com  # Disable host

# ============================================================================
# HOSTGROUP MANAGEMENT
# ============================================================================
ipa hostgroup-find  # List host groups
ipa hostgroup-show webservers  # Show hostgroup details
ipa hostgroup-add webservers --desc="Web Servers"  # Add hostgroup
ipa hostgroup-add-member webservers --hosts=web01.example.com,web02.example.com  # Add hosts
ipa hostgroup-remove-member webservers --hosts=web01.example.com  # Remove host

# ============================================================================
# SERVICE MANAGEMENT
# ============================================================================
ipa service-find  # List all services
ipa service-show HTTP/web01.example.com  # Show service details
ipa service-add HTTP/web01.example.com  # Add service principal
ipa service-del HTTP/web01.example.com  # Delete service
ipa-getkeytab -s ipa.example.com -p HTTP/web01.example.com -k /etc/httpd/http.keytab  # Get service keytab

# ============================================================================
# SUDO RULES
# ============================================================================
ipa sudorule-find  # List sudo rules
ipa sudorule-show admin_all  # Show sudo rule details
ipa sudorule-add webadmin --desc="Web admins sudo"  # Add sudo rule
ipa sudorule-add-user webadmin --users=john  # Add user to rule
ipa sudorule-add-host webadmin --hosts=web01.example.com  # Add host to rule
ipa sudorule-add-allow-command webadmin --sudocmds=/usr/bin/systemctl  # Allow specific command
ipa sudorule-mod webadmin --cmdcat=all  # Allow all commands
ipa sudorule-disable webadmin  # Disable rule
ipa sudorule-enable webadmin  # Enable rule

# ============================================================================
# HBAC (HOST-BASED ACCESS CONTROL)
# ============================================================================
ipa hbacrule-find  # List HBAC rules
ipa hbacrule-show allow_all  # Show HBAC rule
ipa hbacrule-add ssh_access --desc="SSH access rule"  # Add HBAC rule
ipa hbacrule-add-user ssh_access --users=john --groups=developers  # Add users/groups
ipa hbacrule-add-host ssh_access --hosts=web01.example.com  # Add hosts
ipa hbacrule-add-service ssh_access --hbacsvcs=sshd  # Add service
ipa hbacrule-disable allow_all  # Disable rule (disable default allow_all for security)
ipa hbactest --user=john --host=web01.example.com --service=sshd  # Test HBAC rule

# ============================================================================
# DNS MANAGEMENT
# ============================================================================
ipa dnszone-find  # List DNS zones
ipa dnszone-show example.com.  # Show zone details
ipa dnszone-add example.com. --admin-email=admin@example.com  # Add DNS zone
ipa dnsrecord-find example.com.  # List records in zone
ipa dnsrecord-show example.com. www  # Show specific record
ipa dnsrecord-add example.com. www --a-rec=192.168.1.100  # Add A record
ipa dnsrecord-add example.com. mail --a-rec=192.168.1.50 --mx-rec="10 mail"  # Add A + MX
ipa dnsrecord-del example.com. www --a-rec=192.168.1.100  # Delete record

# ============================================================================
# CERTIFICATES
# ============================================================================
ipa cert-request server.csr --principal=HTTP/web01.example.com  # Request certificate
ipa cert-show 1  # Show certificate by serial number
ipa cert-find --subject=web01.example.com  # Find certificates
ipa cert-revoke 1 --revocation-reason=4  # Revoke certificate (reason 4=superseded)
ipa-getcert request -f /etc/pki/tls/certs/server.crt -k /etc/pki/tls/private/server.key -K HTTP/web01.example.com  # Request cert with certmonger
ipa-getcert list  # List tracked certificates (certmonger)

# ============================================================================
# REPLICATION & TOPOLOGY
# ============================================================================
ipa topologysegment-find domain  # Show domain replication topology
ipa topologysegment-find ca  # Show CA replication topology
ipa server-find  # List IPA servers
ipa server-show ipa01.example.com  # Show server details
ipa-replica-manage list  # List replication agreements (legacy)
ipa-csreplica-manage list  # List CA replication agreements (legacy)

# ============================================================================
# TRUSTS (AD INTEGRATION)
# ============================================================================
ipa trust-find  # List AD trusts
ipa trust-show ad.example.com  # Show trust details
ipa trust-add ad.example.com --type=ad --admin=Administrator --password  # Add AD trust

# ============================================================================
# CLIENT ENROLLMENT
# ============================================================================
# On client machine:
ipa-client-install --domain=example.com --server=ipa.example.com --principal=admin  # Enroll client
ipa-client-install --uninstall  # Unenroll client

# ============================================================================
# ADMIN & SERVER MANAGEMENT
# ============================================================================
ipactl status  # Show IPA service status
ipactl start  # Start all IPA services
ipactl stop  # Stop all IPA services
ipactl restart  # Restart all IPA services
ipa-server-install --uninstall  # Uninstall IPA server
ipa config-show  # Show global IPA configuration
ipa config-mod --maxusername=32  # Modify global config

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
ipa ping  # Test IPA connectivity
kinit admin && ipa user-find  # Test authentication
journalctl -u ipa -f  # Watch IPA service logs
journalctl -u krb5kdc -f  # Watch Kerberos logs
ipa-healthcheck  # Run health check (IPA 4.8+)
certutil -L -d /etc/pki/nssdb  # List NSS certificates
klist -e  # Show tickets with encryption types
kvno HTTP/web01.example.com  # Verify service principal

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Add new user:
kinit admin
ipa user-add jsmith --first=John --last=Smith --email=jsmith@example.com
ipa group-add-member developers --users=jsmith
ipa passwd jsmith

# Grant sudo access:
ipa sudorule-add-user webadmin --users=jsmith
ipa sudorule-add-host webadmin --hosts=web01.example.com

# Add new service:
ipa service-add HTTP/app.example.com
ipa-getkeytab -s ipa.example.com -p HTTP/app.example.com -k /etc/httpd/http.keytab

# Create HBAC rule for SSH:
ipa hbacrule-add ssh_devs --desc="Devs SSH access"
ipa hbacrule-add-user ssh_devs --groups=developers
ipa hbacrule-add-host ssh_devs --hostgroups=webservers
ipa hbacrule-add-service ssh_devs --hbacsvcs=sshd

# ============================================================================
# SYNTAX REFERENCE
# ============================================================================
# ipa <object>-<action> [name] [options]
# kinit [username][@REALM]
# ipa-getkeytab -s <server> -p <principal> -k <keytab-file>
# ipactl <status|start|stop|restart>
# ipa-client-install --domain=<domain> --server=<server>
