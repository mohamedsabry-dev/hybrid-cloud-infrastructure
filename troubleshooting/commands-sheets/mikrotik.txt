#!/bin/bash
# ============================================================================
# MIKROTIK ROUTER - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# CONNECTION & BASICS
# ============================================================================
# SSH: ssh admin@192.168.88.1
# Winbox: GUI tool for RouterOS
# WebFig: Web interface http://192.168.88.1

/system identity print  # Show router identity
/system identity set name=MyRouter  # Set router name
/system resource print  # Show system resources (CPU, memory, uptime)
/system routerboard print  # Show hardware info
/system package print  # List installed packages
/system package update check-for-updates  # Check for updates
/system reboot  # Reboot router
/quit  # Exit CLI

# ============================================================================
# INTERFACES
# ============================================================================
/interface print  # List all interfaces
/interface print stats  # Show interface statistics
/interface ethernet print  # Show ethernet interfaces
/interface set ether1 name=WAN  # Rename interface
/interface set ether2 disabled=no  # Enable interface
/interface set ether2 disabled=yes  # Disable interface
/interface monitor ether1  # Monitor interface status

# ============================================================================
# IP ADDRESSING
# ============================================================================
/ip address print  # Show IP addresses
/ip address add address=192.168.1.1/24 interface=ether2  # Add IP address
/ip address remove [find address="192.168.1.1/24"]  # Remove IP address
/ip dhcp-client print  # Show DHCP client status
/ip dhcp-client add interface=ether1 disabled=no  # Enable DHCP client on WAN

# ============================================================================
# DHCP SERVER
# ============================================================================
/ip dhcp-server print  # Show DHCP servers
/ip dhcp-server setup  # Interactive DHCP server setup
/ip dhcp-server add name=lan interface=bridge disabled=no address-pool=lan-pool  # Add DHCP server
/ip pool print  # Show address pools
/ip pool add name=lan-pool ranges=192.168.1.100-192.168.1.200  # Add address pool
/ip dhcp-server network print  # Show DHCP networks
/ip dhcp-server network add address=192.168.1.0/24 gateway=192.168.1.1 dns-server=8.8.8.8,8.8.4.4  # Add DHCP network
/ip dhcp-server lease print  # Show DHCP leases
/ip dhcp-server lease make-static [find address="192.168.1.50"]  # Make lease static

# ============================================================================
# ROUTING
# ============================================================================
/ip route print  # Show routing table
/ip route add dst-address=0.0.0.0/0 gateway=192.168.0.1  # Add default route
/ip route add dst-address=10.0.0.0/8 gateway=192.168.1.254  # Add static route
/ip route remove [find dst-address="10.0.0.0/8"]  # Remove route

# ============================================================================
# FIREWALL
# ============================================================================
/ip firewall filter print  # Show filter rules
/ip firewall filter add chain=input action=accept connection-state=established,related comment="Accept established/related"  # Accept established
/ip firewall filter add chain=input action=drop in-interface=ether1 comment="Drop all from WAN"  # Drop from WAN
/ip firewall filter add chain=forward action=accept connection-state=established,related  # Accept established forward
/ip firewall filter add chain=forward action=drop connection-state=invalid  # Drop invalid
/ip firewall filter move [find comment="Drop all from WAN"] 0  # Move rule to top
/ip firewall filter disable [find comment="Drop all from WAN"]  # Disable rule
/ip firewall filter enable [find comment="Drop all from WAN"]  # Enable rule
/ip firewall filter remove [find comment="Drop all from WAN"]  # Remove rule

# ============================================================================
# NAT
# ============================================================================
/ip firewall nat print  # Show NAT rules
/ip firewall nat add chain=srcnat out-interface=ether1 action=masquerade comment="Masquerade WAN"  # Masquerade (basic NAT)
/ip firewall nat add chain=dstnat dst-port=80 protocol=tcp in-interface=ether1 action=dst-nat to-addresses=192.168.1.10 to-ports=80 comment="HTTP port forward"  # Port forwarding

# ============================================================================
# DNS
# ============================================================================
/ip dns print  # Show DNS settings
/ip dns set servers=8.8.8.8,8.8.4.4 allow-remote-requests=yes  # Set DNS servers
/ip dns cache print  # Show DNS cache
/ip dns cache flush  # Flush DNS cache
/ip dns static print  # Show static DNS entries
/ip dns static add name=router.local address=192.168.1.1  # Add static DNS entry

# ============================================================================
# WIRELESS (IF APPLICABLE)
# ============================================================================
/interface wireless print  # Show wireless interfaces
/interface wireless enable wlan1  # Enable wireless
/interface wireless disable wlan1  # Disable wireless
/interface wireless set wlan1 ssid=MyNetwork mode=ap-bridge  # Configure wireless
/interface wireless security-profiles print  # Show security profiles
/interface wireless security-profiles add name=myprofile authentication-types=wpa2-psk mode=dynamic-keys wpa2-pre-shared-key=mypassword  # Add security profile
/interface wireless set wlan1 security-profile=myprofile  # Apply security profile
/interface wireless registration-table print  # Show connected clients

# ============================================================================
# BRIDGE
# ============================================================================
/interface bridge print  # Show bridges
/interface bridge add name=bridge1  # Create bridge
/interface bridge port print  # Show bridge ports
/interface bridge port add bridge=bridge1 interface=ether2  # Add port to bridge
/interface bridge port add bridge=bridge1 interface=ether3  # Add another port

# ============================================================================
# VLAN
# ============================================================================
/interface vlan print  # Show VLANs
/interface vlan add name=vlan10 vlan-id=10 interface=bridge1  # Add VLAN
/ip address add address=192.168.10.1/24 interface=vlan10  # Add IP to VLAN

# ============================================================================
# VPN - L2TP/IPsec
# ============================================================================
/interface l2tp-server server print  # Show L2TP server status
/interface l2tp-server server set enabled=yes default-profile=default-encryption  # Enable L2TP server
/ppp profile print  # Show PPP profiles
/ppp secret print  # Show VPN users
/ppp secret add name=vpnuser password=vpnpass service=l2tp  # Add VPN user
/interface l2tp-client print  # Show L2TP client connections

# ============================================================================
# VPN - OpenVPN
# ============================================================================
/interface ovpn-server server print  # Show OpenVPN server status
/interface ovpn-server server set enabled=yes  # Enable OpenVPN server
/ppp secret add name=ovpnuser password=ovpnpass service=ovpn  # Add OpenVPN user

# ============================================================================
# BANDWIDTH MONITORING
# ============================================================================
/interface monitor-traffic ether1  # Monitor interface traffic (live)
/tool bandwidth-test 192.168.1.100  # Test bandwidth to host
/queue simple print  # Show simple queues (bandwidth limiting)
/queue simple add name=limit-user target=192.168.1.50/32 max-limit=5M/10M  # Limit user bandwidth (upload/download)

# ============================================================================
# LOGGING & MONITORING
# ============================================================================
/log print  # Show log
/log print follow  # Follow log (like tail -f)
/log print where topics~"system"  # Filter log by topic
/system logging print  # Show logging rules
/tool sniffer print  # Show packet sniffer status
/tool sniffer quick interface=ether1  # Quick packet capture

# ============================================================================
# SYSTEM USERS
# ============================================================================
/user print  # List users
/user add name=newuser password=newpass group=full  # Add user
/user set newuser password=newpassword  # Change password
/user disable newuser  # Disable user
/user remove newuser  # Remove user

# ============================================================================
# BACKUP & RESTORE
# ============================================================================
/system backup save name=backup-$(date +%Y%m%d)  # Create backup
/file print  # List files (including backups)
/file print where name~"backup"  # List backup files
/system backup load name=backup-20240115  # Restore backup
/export file=config-export  # Export configuration to file
/import file-name=config-export.rsc  # Import configuration

# ============================================================================
# SCRIPTS & SCHEDULER
# ============================================================================
/system script print  # Show scripts
/system script add name=myscript source="{ :log info \"Script executed\" }"  # Add script
/system script run myscript  # Run script
/system scheduler print  # Show scheduled tasks
/system scheduler add name=daily-backup on-event="/system backup save name=auto-backup" start-time=02:00:00 interval=1d  # Schedule backup

# ============================================================================
# TIME & NTP
# ============================================================================
/system clock print  # Show system time
/system clock set time-zone-name=America/New_York  # Set timezone
/system ntp client print  # Show NTP client status
/system ntp client set enabled=yes primary-ntp=pool.ntp.org  # Enable NTP client

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
/ping 8.8.8.8  # Ping test
/ping 8.8.8.8 count=10  # Ping 10 times
/tool traceroute 8.8.8.8  # Traceroute
/tool netwatch print  # Show network monitoring
/tool netwatch add host=8.8.8.8 interval=10s  # Monitor host
/ip neighbor print  # Show ARP table
/ip arp print  # Show ARP cache
/interface ethernet monitor ether1  # Monitor ethernet status
/tool bandwidth-test 192.168.1.100 protocol=tcp  # Bandwidth test

# ============================================================================
# RESET & RECOVERY
# ============================================================================
/system reset-configuration  # Reset to defaults (confirmation required)
/system reset-configuration no-defaults=yes skip-backup=yes  # Hard reset
# Physical reset: Hold reset button for 5 seconds during boot

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Basic router setup:
/ip address add address=192.168.1.1/24 interface=ether2
/ip pool add name=dhcp-pool ranges=192.168.1.100-192.168.1.200
/ip dhcp-server add name=dhcp1 interface=ether2 address-pool=dhcp-pool disabled=no
/ip dhcp-server network add address=192.168.1.0/24 gateway=192.168.1.1 dns-server=8.8.8.8
/ip firewall nat add chain=srcnat out-interface=ether1 action=masquerade

# Port forwarding:
/ip firewall nat add chain=dstnat dst-port=22 protocol=tcp in-interface=ether1 action=dst-nat to-addresses=192.168.1.10 to-ports=22

# Backup configuration:
/system backup save name=backup-$(date +%Y%m%d)
/export file=config-export

# Block IP address:
/ip firewall filter add chain=forward src-address=192.168.1.50 action=drop

# ============================================================================
# SYNTAX REFERENCE
# ============================================================================
# /<menu> <command> [parameters]
# /ip address print  # Show items
# /ip address add <key=value>  # Add item
# /ip address set <number> <key=value>  # Modify item
# /ip address remove <number>  # Remove item
# /ip address enable <number>  # Enable item
# /ip address disable <number>  # Disable item
# [find <key=value>]  # Find item by criteria
# Tab key: Auto-complete
# ?: Show help
# .. : Go up one menu level
