#!/bin/bash
# ============================================================================
# WORDPRESS - Essential Operations & Logs (Max 70 Lines)
# ============================================================================

# ============================================================================
# CORE OPERATIONS
# ============================================================================
wp core version  # Show WordPress version
wp core update  # Update WordPress core
wp core update-db  # Update database after core update

# ============================================================================
# PLUGINS & THEMES
# ============================================================================
wp plugin list  # List all plugins
wp plugin update --all  # Update all plugins
wp plugin activate <plugin-name>  # Activate plugin
wp plugin deactivate --all  # Deactivate all (troubleshooting)
wp theme list  # List all themes
wp theme update --all  # Update all themes

# ============================================================================
# DATABASE OPERATIONS
# ============================================================================
wp db export backup.sql  # Export database
wp db import backup.sql  # Import database
wp db optimize  # Optimize database
wp db repair  # Repair database
wp db check  # Check database connection
wp db size --human-readable  # Show database size

# ============================================================================
# SEARCH & REPLACE (MIGRATION)
# ============================================================================
wp search-replace 'http://old-site.com' 'https://new-site.com' --dry-run  # Test first
wp search-replace 'http://old-site.com' 'https://new-site.com'  # Execute replacement

# ============================================================================
# CACHE & MAINTENANCE
# ============================================================================
wp cache flush  # Flush object cache
wp transient delete --all  # Delete transients
wp rewrite flush  # Flush rewrite rules
wp cron event run --due-now  # Run due cron jobs

# ============================================================================
# USER MANAGEMENT
# ============================================================================
wp user list  # List all users
wp user create john john@example.com --role=editor  # Create user
wp user update 1 --user_pass=newpassword  # Reset password

# ============================================================================
# LOG LOCATIONS
# ============================================================================
# WordPress debug log:
tail -f /var/www/html/wp-content/debug.log

# PHP error log (Debian/Ubuntu):
tail -f /var/log/apache2/error.log
tail -f /var/log/php8.1-fpm.log

# PHP error log (RHEL/CentOS):
tail -f /var/log/httpd/error_log
tail -f /var/log/php-fpm/www-error.log

# NGINX error log:
tail -f /var/log/nginx/error.log

# Enable debug logging (add to wp-config.php):
# define( 'WP_DEBUG', true );
# define( 'WP_DEBUG_LOG', true );
# define( 'WP_DEBUG_DISPLAY', false );

# ============================================================================
# FILE PERMISSIONS
# ============================================================================
find /var/www/html -type d -exec chmod 755 {} \;  # Directories
find /var/www/html -type f -exec chmod 644 {} \;  # Files
chmod 600 /var/www/html/wp-config.php  # wp-config.php
chown -R www-data:www-data /var/www/html  # Ownership (Debian/Ubuntu)

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
wp plugin deactivate --all  # Disable all plugins (white screen fix)
wp theme activate twentytwentythree  # Switch to default theme
rm /var/www/html/.maintenance  # Remove stuck maintenance mode
wp config get DB_NAME  # Verify database settings
