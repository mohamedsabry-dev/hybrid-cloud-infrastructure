# ============================================================================
# SERVICE MANAGEMENT
# ============================================================================
systemctl status nginx  # Check NGINX status
systemctl start nginx  # Start NGINX
systemctl stop nginx  # Stop NGINX
systemctl restart nginx  # Restart NGINX
systemctl reload nginx  # Reload config without dropping connections
systemctl enable nginx  # Enable at boot
nginx -v  # Show version
nginx -V  # Show version with compile options

# ============================================================================
# CONFIGURATION TESTING
# ============================================================================
nginx -t  # Test configuration syntax
nginx -T  # Test and dump configuration
nginx -t -c /etc/nginx/nginx.conf  # Test specific config file
nginx -s reload  # Reload config (alternative to systemctl reload)
nginx -s quit  # Graceful shutdown
nginx -s stop  # Fast shutdown

# ============================================================================
# CONFIGURATION FILES
# ============================================================================
# Main config: /etc/nginx/nginx.conf
# Site configs: /etc/nginx/sites-available/
# Enabled sites: /etc/nginx/sites-enabled/
# Modules: /etc/nginx/modules-enabled/
# Logs: /var/log/nginx/

# Enable site (Debian/Ubuntu):
ln -s /etc/nginx/sites-available/mysite /etc/nginx/sites-enabled/

# Disable site:
rm /etc/nginx/sites-enabled/mysite

# ============================================================================
# LOGS & MONITORING
# ============================================================================
tail -f /var/log/nginx/access.log  # Follow access log
tail -f /var/log/nginx/error.log  # Follow error log
journalctl -u nginx -f  # Follow systemd logs
grep "HTTP/1.1\" 404" /var/log/nginx/access.log  # Find 404 errors
grep "error" /var/log/nginx/error.log  # Find errors
awk '{print $1}' /var/log/nginx/access.log | sort | uniq -c | sort -rn | head -10  # Top 10 IPs

# DEBUGGING & TROUBLESHOOTING
# ============================================================================
# Enable debug log (in http or server block):
# error_log /var/log/nginx/debug.log debug;

# Check which config file is being used:
nginx -V 2>&1 | grep -o '\-\-conf-path=\S*'

# Check listening ports:
ss -tulpn | grep nginx

# Check NGINX processes:
ps aux | grep nginx

# Check NGINX master process PID:
cat /var/run/nginx.pid

# Send signal to master process:
kill -HUP $(cat /var/run/nginx.pid)  # Reload config
kill -USR1 $(cat /var/run/nginx.pid)  # Reopen log files
kill -QUIT $(cat /var/run/nginx.pid)  # Graceful shutdown

# ============================================================================
# 502 Bad Gateway:
# - Check upstream service is running
# - Check firewall/SELinux not blocking
# - Check proxy_pass URL is correct

# 403 Forbidden:
# - Check file permissions
# - Check NGINX user has access
# - Check directory index is set
# - Check SELinux context (if RHEL/CentOS)

# 404 Not Found:
# - Check root directive
# - Check try_files directive
# - Check file exists on disk

# Config test fails:
nginx -t  # See specific error
# Check syntax, missing semicolons, brackets

# SELinux issues (RHEL/CentOS):
getsebool -a | grep httpd  # Check booleans
setsebool -P httpd_can_network_connect 1  # Allow proxy
setenforce 0  # Temporarily disable (testing only)

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Add new site:

ln -s /etc/nginx/sites-available/newsite /etc/nginx/sites-enabled/
nginx -t
systemctl reload nginx

# Update configuration:
vim /etc/nginx/sites-available/mysite
nginx -t
systemctl reload nginx

# Emergency reload:
nginx -s reload

# Rotate logs manually:
mv /var/log/nginx/access.log /var/log/nginx/access.log.1
nginx -s reopen
