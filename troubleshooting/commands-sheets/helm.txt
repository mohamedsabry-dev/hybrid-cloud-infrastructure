#!/bin/bash
# ============================================================================
# HELM - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# REPOSITORY MANAGEMENT
# ============================================================================
helm repo add bitnami https://charts.bitnami.com/bitnami  # Add repository
helm repo list  # List repositories
helm repo update  # Update repository index (like apt update)
helm repo remove bitnami  # Remove repository
helm search repo nginx  # Search repositories
helm search repo nginx --versions  # Show all available versions
helm search hub wordpress  # Search Helm Hub (public charts)

# ============================================================================
# CHART INFORMATION
# ============================================================================
helm show chart bitnami/nginx  # Show chart metadata
helm show values bitnami/nginx  # Show default values.yaml
helm show readme bitnami/nginx  # Show chart README
helm show all bitnami/nginx  # Show everything

# ============================================================================
# INSTALLING RELEASES
# ============================================================================
helm install my-nginx bitnami/nginx  # Install with release name
helm install my-nginx bitnami/nginx -n my-namespace  # Specific namespace
helm install my-nginx bitnami/nginx --create-namespace -n my-namespace  # Create namespace if missing
helm install my-nginx bitnami/nginx --set replicaCount=3  # Override single value
helm install my-nginx bitnami/nginx --set service.type=NodePort --set service.nodePort=30080  # Multiple values
helm install my-nginx bitnami/nginx -f custom-values.yaml  # Values from file
helm install my-nginx bitnami/nginx -f values1.yaml -f values2.yaml  # Multiple files (later overrides)
helm install my-nginx bitnami/nginx --version 13.2.0  # Specific chart version
helm install my-nginx ./my-chart/  # Install local chart
helm install my-nginx bitnami/nginx --dry-run --debug  # Simulate install (show manifests)
helm install my-nginx bitnami/nginx --wait --timeout 5m  # Wait for resources to be ready

# ============================================================================
# LISTING & STATUS
# ============================================================================
helm list  # List releases in current namespace
helm list -A  # List releases in all namespaces
helm list -n my-namespace  # Specific namespace
helm list --all  # Include uninstalled releases
helm status my-nginx  # Show release status
helm history my-nginx  # Show release revision history

# ============================================================================
# UPGRADING RELEASES
# ============================================================================
helm upgrade my-nginx bitnami/nginx  # Upgrade to latest chart version
helm upgrade my-nginx bitnami/nginx --set replicaCount=5  # Upgrade with new values
helm upgrade my-nginx bitnami/nginx -f new-values.yaml  # Upgrade with values file
helm upgrade --install my-nginx bitnami/nginx  # Install if not exists, upgrade if exists
helm upgrade my-nginx bitnami/nginx --version 14.0.0  # Upgrade to specific version
helm upgrade my-nginx bitnami/nginx --reuse-values  # Keep previous values
helm upgrade my-nginx bitnami/nginx --reset-values  # Reset to chart defaults
helm upgrade my-nginx bitnami/nginx --wait --timeout 10m  # Wait for completion
helm upgrade my-nginx bitnami/nginx --atomic  # Rollback automatically on failure
helm upgrade my-nginx bitnami/nginx --dry-run --debug  # Simulate upgrade

# ============================================================================
# ROLLBACK
# ============================================================================
helm rollback my-nginx  # Rollback to previous revision
helm rollback my-nginx 2  # Rollback to specific revision
helm rollback my-nginx 2 --wait --timeout 5m  # Rollback with wait

# ============================================================================
# UNINSTALLING
# ============================================================================
helm uninstall my-nginx  # Uninstall release
helm uninstall my-nginx -n my-namespace  # Specific namespace
helm uninstall my-nginx --keep-history  # Uninstall but keep history

# ============================================================================
# GETTING RELEASE INFO
# ============================================================================
helm get values my-nginx  # Get user-supplied values
helm get values my-nginx --all  # Get all values (including defaults)
helm get manifest my-nginx  # Get rendered Kubernetes manifests
helm get notes my-nginx  # Get release notes
helm get all my-nginx  # Get everything

# ============================================================================
# TEMPLATE RENDERING & VALIDATION
# ============================================================================
helm template my-nginx bitnami/nginx  # Render templates locally (no install)
helm template my-nginx bitnami/nginx -f values.yaml  # Render with values
helm template my-nginx bitnami/nginx --set replicaCount=3  # Render with overrides
helm template my-nginx ./my-chart --show-only templates/deployment.yaml  # Render specific template
helm lint ./my-chart/  # Validate chart for issues
helm lint ./my-chart/ -f values.yaml  # Lint with values

# ============================================================================
# CHART MANAGEMENT
# ============================================================================
helm pull bitnami/nginx  # Download chart
helm pull bitnami/nginx --version 13.2.0  # Download specific version
helm pull bitnami/nginx --untar  # Download and extract
helm create my-chart  # Create new chart skeleton
helm package ./my-chart/  # Package chart directory to .tgz
helm package ./my-chart/ --version 1.0.0  # Package with version

# ============================================================================
# DEPENDENCIES
# ============================================================================
helm dependency update ./my-chart/  # Download dependencies to charts/
helm dependency list ./my-chart/  # List chart dependencies
# Chart.yaml dependencies example:
# dependencies:
#   - name: postgresql
#     version: "11.x.x"
#     repository: https://charts.bitnami.com/bitnami

# ============================================================================
# TESTING
# ============================================================================
helm test my-nginx  # Run tests defined in chart
helm test my-nginx --logs  # Show test pod logs

# ============================================================================
# PLUGINS (OPTIONAL)
# ============================================================================
helm plugin list  # List installed plugins
helm plugin install https://github.com/databus23/helm-diff  # Install helm-diff plugin
helm diff upgrade my-nginx bitnami/nginx -f new-values.yaml  # Show diff before upgrade (requires plugin)

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
helm install my-nginx bitnami/nginx --dry-run --debug  # Debug installation
helm status my-nginx  # Check release status
helm history my-nginx  # Check revision history
helm get manifest my-nginx  # View rendered manifests
helm get values my-nginx --all  # View all values
helm lint ./my-chart/  # Validate chart
helm template my-nginx ./my-chart/ --debug  # Debug template rendering
kubectl get all -l app.kubernetes.io/instance=my-nginx  # Get Kubernetes resources for release

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Initial setup:
helm repo add bitnami https://charts.bitnami.com/bitnami
helm repo update
helm show values bitnami/nginx > values.yaml
# Edit values.yaml
helm install my-nginx bitnami/nginx -f values.yaml

# Safe upgrade workflow:
helm diff upgrade my-nginx bitnami/nginx -f values.yaml  # 1. Show diff (requires plugin)
helm upgrade my-nginx bitnami/nginx -f values.yaml --dry-run --debug  # 2. Dry-run
helm upgrade my-nginx bitnami/nginx -f values.yaml --atomic --wait  # 3. Upgrade with auto-rollback

# Emergency rollback:
helm history my-nginx  # Check revisions
helm rollback my-nginx 2  # Rollback to working revision
