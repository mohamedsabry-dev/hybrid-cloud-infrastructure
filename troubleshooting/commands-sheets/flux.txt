#!/bin/bash
# ============================================================================
# FLUX (GitOps) - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# FLUX INSTALLATION & BOOTSTRAP
# ============================================================================
flux check --pre  # Check prerequisites
flux install  # Install Flux on cluster
flux bootstrap github --owner=myorg --repository=fleet-infra --path=clusters/production --personal  # Bootstrap with GitHub
flux bootstrap gitlab --owner=myorg --repository=fleet-infra --path=clusters/production  # Bootstrap with GitLab
flux uninstall  # Uninstall Flux

# ============================================================================
# FLUX STATUS & VERSION
# ============================================================================
flux version  # Show Flux version
flux check  # Check Flux installation status
flux get all  # Show all Flux resources
flux get sources all  # Show all sources (GitRepository, HelmRepository)
flux get kustomizations  # Show Kustomizations
flux get helmreleases  # Show Helm releases

# ============================================================================
# GIT SOURCES
# ============================================================================
flux create source git my-app --url=https://github.com/myorg/my-app --branch=main --interval=1m  # Create GitRepository
flux get sources git  # List Git sources
flux get sources git my-app  # Show specific Git source
flux delete source git my-app  # Delete Git source
flux reconcile source git my-app  # Force reconcile (pull latest)
flux suspend source git my-app  # Suspend reconciliation
flux resume source git my-app  # Resume reconciliation

# ============================================================================
# KUSTOMIZATIONS
# ============================================================================
flux create kustomization my-app --source=GitRepository/my-app --path=./deploy --prune=true --interval=5m  # Create Kustomization
flux get kustomizations  # List Kustomizations
flux get kustomization my-app  # Show specific Kustomization
flux delete kustomization my-app  # Delete Kustomization
flux reconcile kustomization my-app --with-source  # Force reconcile with source update
flux suspend kustomization my-app  # Suspend reconciliation
flux resume kustomization my-app  # Resume reconciliation

# ============================================================================
# HELM REPOSITORIES
# ============================================================================
flux create source helm bitnami --url=https://charts.bitnami.com/bitnami --interval=1h  # Add Helm repo
flux get sources helm  # List Helm repositories
flux reconcile source helm bitnami  # Update Helm repo index

# ============================================================================
# HELM RELEASES
# ============================================================================
flux create helmrelease nginx --source=HelmRepository/bitnami --chart=nginx --interval=5m  # Create Helm release
flux get helmreleases  # List Helm releases
flux get helmrelease nginx  # Show specific Helm release
flux reconcile helmrelease nginx  # Force Helm upgrade
flux suspend helmrelease nginx  # Suspend Helm release
flux resume helmrelease nginx  # Resume Helm release

# ============================================================================
# LOGS & TROUBLESHOOTING
# ============================================================================
flux logs  # Show Flux controller logs
flux logs --all-namespaces  # Show logs from all namespaces
flux logs --kind=Kustomization --name=my-app  # Show logs for specific Kustomization
flux events --for Kustomization/my-app  # Show events for resource
kubectl logs -n flux-system deploy/source-controller  # Source controller logs
kubectl logs -n flux-system deploy/kustomize-controller  # Kustomize controller logs
kubectl logs -n flux-system deploy/helm-controller  # Helm controller logs
kubectl logs -n flux-system deploy/notification-controller  # Notification controller logs

# ============================================================================
# RECONCILIATION & SYNC
# ============================================================================
flux reconcile source git flux-system  # Sync with Git repository
flux reconcile kustomization flux-system  # Apply changes from Git
flux reconcile helmrelease my-app  # Update Helm release
flux reconcile source git my-app --with-source  # Sync Git source first

# ============================================================================
# EXPORT & IMPORT
# ============================================================================
flux export source git my-app > my-app-source.yaml  # Export GitRepository manifest
flux export kustomization my-app > my-app-kustomization.yaml  # Export Kustomization manifest
flux export helmrelease my-app > my-app-helmrelease.yaml  # Export HelmRelease manifest

# ============================================================================
# IMAGE AUTOMATION (OPTIONAL)
# ============================================================================
flux get image repository  # List image repositories
flux get image policy  # List image update policies
flux reconcile image repository my-app  # Scan for new images

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Check Flux health:
flux check

# Force full sync:
flux reconcile source git flux-system --with-source
flux reconcile kustomization flux-system

# Debug Kustomization not applying:
flux get kustomization my-app
flux logs --kind=Kustomization --name=my-app
kubectl get events -n flux-system --sort-by='.lastTimestamp'

# Add new application to Flux:
flux create source git my-new-app --url=https://github.com/myorg/my-new-app --branch=main
flux create kustomization my-new-app --source=GitRepository/my-new-app --path=./k8s --prune=true

# ============================================================================
# FLUX COMPONENTS (NAMESPACE: flux-system)
# ============================================================================
kubectl get pods -n flux-system  # Show Flux pods
kubectl get all -n flux-system  # Show all Flux resources
kubectl describe kustomization my-app -n flux-system  # Describe Kustomization
kubectl describe gitrepository flux-system -n flux-system  # Describe GitRepository

# ============================================================================
# SYNTAX REFERENCE
# ============================================================================
# flux get <resource-type> [name]
# flux create <resource-type> <name> [options]
# flux reconcile <resource-type> <name> [--with-source]
# flux suspend <resource-type> <name>
# flux resume <resource-type> <name>
# flux delete <resource-type> <name>
# flux logs [--kind=<type>] [--name=<name>]
