#!/bin/bash
# ============================================================================
# GITHUB ACTIONS RUNNER - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# RUNNER INSTALLATION (LINUX/MAC)
# ============================================================================
# Download runner (check latest version at https://github.com/actions/runner/releases):
mkdir actions-runner && cd actions-runner
curl -o actions-runner-linux-x64-2.311.0.tar.gz -L https://github.com/actions/runner/releases/download/v2.311.0/actions-runner-linux-x64-2.311.0.tar.gz
tar xzf actions-runner-linux-x64-2.311.0.tar.gz

# Configure runner:
./config.sh --url https://github.com/OWNER/REPO --token YOUR_TOKEN  # Repository runner
./config.sh --url https://github.com/ORGANIZATION --token YOUR_TOKEN  # Organization runner
./config.sh --url https://github.com/enterprises/ENTERPRISE --token YOUR_TOKEN  # Enterprise runner

# Configure with labels:
./config.sh --url https://github.com/OWNER/REPO --token YOUR_TOKEN --labels linux,x64,gpu

# Configure as service:
sudo ./svc.sh install  # Install as systemd service
sudo ./svc.sh start  # Start service
sudo ./svc.sh status  # Check status
sudo ./svc.sh stop  # Stop service
sudo ./svc.sh uninstall  # Uninstall service

# Run interactively (testing):
./run.sh  # Run runner in foreground

# ============================================================================
# RUNNER MANAGEMENT (SERVICE)
# ============================================================================
sudo systemctl status actions.runner.*  # Check runner service status
sudo systemctl start actions.runner.*  # Start runner service
sudo systemctl stop actions.runner.*  # Stop runner service
sudo systemctl restart actions.runner.*  # Restart runner service
sudo systemctl enable actions.runner.*  # Enable at boot
sudo journalctl -u actions.runner.* -f  # Follow runner logs

# ============================================================================
# RUNNER REMOVAL
# ============================================================================
sudo ./svc.sh stop  # Stop service
sudo ./svc.sh uninstall  # Uninstall service
./config.sh remove --token YOUR_TOKEN  # Remove runner from GitHub
rm -rf ~/actions-runner  # Delete runner directory


# ============================================================================
# MONITORING & LOGS
# ============================================================================
tail -f _diag/Runner_*.log  # Follow runner diagnostic logs
tail -f _diag/Worker_*.log  # Follow worker logs
ls -lh _diag/  # List diagnostic logs
journalctl -u actions.runner.* -n 100  # Last 100 log lines
journalctl -u actions.runner.* --since "1 hour ago"  # Logs from last hour


# ============================================================================
# RUNNER UPDATES
# ============================================================================
# Runners auto-update when jobs run
# Manual update:
sudo ./svc.sh stop
./config.sh remove --token YOUR_TOKEN
# Download new runner version
./config.sh --url https://github.com/OWNER/REPO --token YOUR_TOKEN
sudo ./svc.sh install
sudo ./svc.sh start

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
# Runner offline:
sudo systemctl status actions.runner.*  # Check service status
journalctl -u actions.runner.* -n 50  # Check recent logs
./run.sh  # Test run interactively
ping github.com  # Check network connectivity

# Runner not picking up jobs:
# - Check runner is online in GitHub UI
# - Check labels match workflow requirements
# - Check runner group access (if using groups)
# - Restart runner service

# Permission errors:
sudo chown -R runner:runner /home/runner/actions-runner  # Fix ownership
chmod +x config.sh run.sh  # Make scripts executable

# Disk space issues:
df -h  # Check disk space
du -sh _work/*  # Check workspace size
rm -rf _work/*  # Clean workspaces (BE CAREFUL)

# ============================================================================
# MACOS SPECIFIC
# ============================================================================
# Install as LaunchAgent (macOS):
./svc.sh install  # Install service
./svc.sh start  # Start service
launchctl list | grep actions.runner  # Check service
tail -f ~/Library/Logs/actions.runner.*  # View logs

# Uninstall (macOS):
./svc.sh stop
./svc.sh uninstall

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Setup new runner:
mkdir ~/actions-runner && cd ~/actions-runner
curl -o actions-runner.tar.gz -L https://github.com/actions/runner/releases/latest/download/actions-runner-linux-x64-*.tar.gz
tar xzf actions-runner.tar.gz
./config.sh --url https://github.com/OWNER/REPO --token TOKEN --labels production,linux
sudo ./svc.sh install
sudo ./svc.sh start

# Check runner health:
sudo systemctl status actions.runner.*
journalctl -u actions.runner.* -n 20
tail -f _diag/Runner_*.log

# Clean runner workspace:
sudo ./svc.sh stop
rm -rf _work/*
sudo ./svc.sh start

# Upgrade runner:
sudo ./svc.sh stop
./config.sh remove --token TOKEN
# Download new version
./config.sh --url https://github.com/OWNER/REPO --token TOKEN
sudo ./svc.sh install
sudo ./svc.sh start

# ============================================================================
# GITHUB CLI (gh) - RUNNER MANAGEMENT
# ============================================================================
gh api repos/OWNER/REPO/actions/runners  # List runners (API)
gh api repos/OWNER/REPO/actions/runners/RUNNER_ID  # Get runner details
gh api --method DELETE repos/OWNER/REPO/actions/runners/RUNNER_ID  # Remove runner

# ============================================================================
# SYNTAX REFERENCE
# ============================================================================
# ./config.sh --url <repo-url> --token <token> [--labels <labels>]
# ./run.sh  # Run interactively
# sudo ./svc.sh <install|start|stop|status|uninstall>
# journalctl -u actions.runner.* [-f|-n N]
# runs-on: [self-hosted, <label1>, <label2>]  # Workflow syntax
