#!/bin/bash
# ============================================================================
# MARIADB/MYSQL - Essential Operations, Logs & Health (Max 70 Lines)
# ============================================================================

# ============================================================================
# CONNECTION
# ============================================================================
mysql -u root -p  # Connect as root
mysql -u user -p database_name  # Connect to specific database
mysql -h hostname -u user -p  # Connect to remote host
mysqladmin -u root -p status  # Show server status

# ============================================================================
# DATABASE OPERATIONS
# ============================================================================
SHOW DATABASES;  # List databases
CREATE DATABASE mydb CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;  # Create database
USE mydb;  # Switch to database
DROP DATABASE mydb;  # Delete database

# ============================================================================
# TABLE OPERATIONS
# ============================================================================
SHOW TABLES;  # List tables
DESCRIBE users;  # Show table structure
SELECT * FROM users LIMIT 10;  # Select rows
SELECT COUNT(*) FROM users;  # Count rows

# ============================================================================
# USER MANAGEMENT
# ============================================================================
SELECT User, Host FROM mysql.user;  # List users
CREATE USER 'appuser'@'localhost' IDENTIFIED BY 'password';  # Create user
GRANT ALL PRIVILEGES ON mydb.* TO 'appuser'@'localhost';  # Grant privileges
SHOW GRANTS FOR 'user'@'localhost';  # Show user privileges
FLUSH PRIVILEGES;  # Reload grant tables
ALTER USER 'user'@'localhost' IDENTIFIED BY 'newpassword';  # Change password

# ============================================================================
# BACKUP & RESTORE
# ============================================================================
mysqldump -u root -p mydb > mydb_backup.sql  # Backup database
mysqldump -u root -p --all-databases > all_databases.sql  # Backup all
mysql -u root -p mydb < mydb_backup.sql  # Restore database
mysqldump -u root -p mydb | gzip > mydb_backup.sql.gz  # Backup with compression
gunzip < mydb_backup.sql.gz | mysql -u root -p mydb  # Restore from compressed

# ============================================================================
# DATABASE HEALTH & MONITORING
# ============================================================================
SHOW STATUS;  # Show server status variables
SHOW STATUS LIKE 'Threads%';  # Show thread status
SHOW VARIABLES;  # Show server variables
SHOW VARIABLES LIKE 'max_connections';  # Show specific variable
SHOW PROCESSLIST;  # Show running queries
SHOW FULL PROCESSLIST;  # Show full query text
KILL 123;  # Kill query by process ID
SHOW TABLE STATUS;  # Show table statistics
ANALYZE TABLE users;  # Update table statistics
OPTIMIZE TABLE users;  # Optimize table
CHECK TABLE users;  # Check table for errors
REPAIR TABLE users;  # Repair corrupted table

# ============================================================================
# LOG LOCATIONS
# ============================================================================
# Error log (check my.cnf for location):
tail -f /var/log/mysql/error.log  # Debian/Ubuntu
tail -f /var/log/mariadb/mariadb.log  # RHEL/CentOS

# Slow query log:
SHOW VARIABLES LIKE 'slow_query_log_file';  # Show log location
tail -f /var/log/mysql/slow.log

# General query log (not recommended for production):
SHOW VARIABLES LIKE 'general_log_file';

# Binary logs:
SHOW BINARY LOGS;  # List binary logs

# ============================================================================
# SERVICE MANAGEMENT
# ============================================================================
systemctl status mariadb  # Check service status
systemctl restart mariadb  # Restart service
journalctl -u mariadb -f  # Follow service logs
