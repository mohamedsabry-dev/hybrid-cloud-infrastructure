#!/bin/bash
# ============================================================================
# MARIADB/MYSQL - Production Daily Commands (Pattern Reference)
# ============================================================================

# ============================================================================
# CONNECTION & BASIC OPERATIONS
# ============================================================================
mysql -u root -p  # Connect as root with password prompt
mysql -u user -p database_name  # Connect to specific database
mysql -h hostname -u user -p  # Connect to remote host
mysql -u user -p -e "SELECT * FROM users;"  # Execute query from command line
mysqladmin -u root -p status  # Show server status
mysqladmin -u root -p version  # Show server version

# ============================================================================
# DATABASE MANAGEMENT
# ============================================================================
SHOW DATABASES;  # List all databases
CREATE DATABASE mydb;  # Create database
CREATE DATABASE IF NOT EXISTS mydb CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;  # Create with charset
USE mydb;  # Switch to database
DROP DATABASE mydb;  # Delete database
SELECT DATABASE();  # Show current database

# ============================================================================
# TABLE MANAGEMENT
# ============================================================================
SHOW TABLES;  # List tables in current database
DESCRIBE users;  # Show table structure
SHOW CREATE TABLE users;  # Show CREATE statement
CREATE TABLE users (id INT AUTO_INCREMENT PRIMARY KEY, username VARCHAR(50), email VARCHAR(100));  # Create table
DROP TABLE users;  # Delete table
TRUNCATE TABLE users;  # Empty table (fast)
ALTER TABLE users ADD COLUMN phone VARCHAR(20);  # Add column
ALTER TABLE users MODIFY COLUMN phone VARCHAR(30);  # Modify column
ALTER TABLE users DROP COLUMN phone;  # Drop column
RENAME TABLE old_name TO new_name;  # Rename table

# ============================================================================
# DATA OPERATIONS (CRUD)
# ============================================================================
SELECT * FROM users;  # Select all rows
SELECT username, email FROM users WHERE id = 1;  # Select specific columns/rows
SELECT * FROM users WHERE username LIKE 'john%';  # Pattern matching
SELECT * FROM users ORDER BY id DESC LIMIT 10;  # Sort and limit
INSERT INTO users (username, email) VALUES ('john', 'john@example.com');  # Insert row
INSERT INTO users (username, email) VALUES ('jane', 'jane@example.com'), ('bob', 'bob@example.com');  # Insert multiple
UPDATE users SET email = 'newemail@example.com' WHERE id = 1;  # Update row
DELETE FROM users WHERE id = 1;  # Delete row
SELECT COUNT(*) FROM users;  # Count rows

# ============================================================================
# USER MANAGEMENT
# ============================================================================
SELECT User, Host FROM mysql.user;  # List users
CREATE USER 'newuser'@'localhost' IDENTIFIED BY 'password';  # Create user
CREATE USER 'appuser'@'%' IDENTIFIED BY 'password';  # Create user (any host)
GRANT ALL PRIVILEGES ON mydb.* TO 'newuser'@'localhost';  # Grant all privileges on database
GRANT SELECT, INSERT, UPDATE ON mydb.users TO 'newuser'@'localhost';  # Grant specific privileges
GRANT ALL PRIVILEGES ON *.* TO 'admin'@'localhost' WITH GRANT OPTION;  # Grant all with grant option
REVOKE ALL PRIVILEGES ON mydb.* FROM 'newuser'@'localhost';  # Revoke privileges
DROP USER 'newuser'@'localhost';  # Delete user
SET PASSWORD FOR 'user'@'localhost' = PASSWORD('newpassword');  # Change password
ALTER USER 'user'@'localhost' IDENTIFIED BY 'newpassword';  # Change password (modern)
FLUSH PRIVILEGES;  # Reload grant tables
SHOW GRANTS FOR 'user'@'localhost';  # Show user privileges

# ============================================================================
# BACKUP & RESTORE
# ============================================================================
# Backup single database:
mysqldump -u root -p mydb > mydb_backup.sql

# Backup all databases:
mysqldump -u root -p --all-databases > all_databases.sql

# Backup with compression:
mysqldump -u root -p mydb | gzip > mydb_backup.sql.gz

# Backup specific tables:
mysqldump -u root -p mydb users posts > tables_backup.sql

# Backup structure only (no data):
mysqldump -u root -p --no-data mydb > mydb_structure.sql

# Restore database:
mysql -u root -p mydb < mydb_backup.sql

# Restore from compressed backup:
gunzip < mydb_backup.sql.gz | mysql -u root -p mydb

# Restore all databases:
mysql -u root -p < all_databases.sql

# ============================================================================
# INDEXES
# ============================================================================
SHOW INDEX FROM users;  # Show indexes
CREATE INDEX idx_username ON users(username);  # Create index
CREATE UNIQUE INDEX idx_email ON users(email);  # Create unique index
DROP INDEX idx_username ON users;  # Drop index
ALTER TABLE users ADD INDEX idx_email (email);  # Add index via ALTER

# ============================================================================
# PERFORMANCE & MONITORING
# ============================================================================
SHOW PROCESSLIST;  # Show running queries
SHOW FULL PROCESSLIST;  # Show full query text
KILL 123;  # Kill query by process ID
SHOW STATUS;  # Show server status
SHOW STATUS LIKE 'Threads%';  # Show thread status
SHOW VARIABLES;  # Show server variables
SHOW VARIABLES LIKE 'max_connections';  # Show specific variable
SET GLOBAL max_connections = 200;  # Change global variable
SHOW TABLE STATUS;  # Show table statistics
ANALYZE TABLE users;  # Update table statistics
OPTIMIZE TABLE users;  # Defragment and optimize table
EXPLAIN SELECT * FROM users WHERE username = 'john';  # Explain query execution plan

# ============================================================================
# SLOW QUERY LOG
# ============================================================================
SHOW VARIABLES LIKE 'slow_query%';  # Show slow query log settings
SET GLOBAL slow_query_log = 'ON';  # Enable slow query log
SET GLOBAL long_query_time = 2;  # Set slow query threshold (seconds)
SHOW VARIABLES LIKE 'slow_query_log_file';  # Show log file location

# ============================================================================
# REPLICATION (MASTER-SLAVE)
# ============================================================================
# Master:
SHOW MASTER STATUS;  # Show master status
GRANT REPLICATION SLAVE ON *.* TO 'replica'@'%' IDENTIFIED BY 'password';

# Slave:
SHOW SLAVE STATUS\G  # Show slave status
CHANGE MASTER TO MASTER_HOST='192.168.1.10', MASTER_USER='replica', MASTER_PASSWORD='password', MASTER_LOG_FILE='mysql-bin.000001', MASTER_LOG_POS=107;
START SLAVE;  # Start replication
STOP SLAVE;  # Stop replication

# ============================================================================
# BINARY LOGS
# ============================================================================
SHOW BINARY LOGS;  # List binary logs
SHOW MASTER LOGS;  # Same as above
PURGE BINARY LOGS TO 'mysql-bin.000010';  # Purge old binary logs
PURGE BINARY LOGS BEFORE '2024-01-01 00:00:00';  # Purge by date
FLUSH LOGS;  # Rotate binary log

# ============================================================================
# TRANSACTIONS
# ============================================================================
START TRANSACTION;  # Begin transaction
# ... run queries ...
COMMIT;  # Commit transaction
ROLLBACK;  # Rollback transaction
SET autocommit = 0;  # Disable auto-commit
SET autocommit = 1;  # Enable auto-commit

# ============================================================================
# WORDPRESS SPECIFIC
# ============================================================================
# WordPress database name is typically: wordpress or wp_*
# WordPress tables prefix: wp_ (default)

# Change WordPress URL:
UPDATE wp_options SET option_value = 'https://newdomain.com' WHERE option_name IN ('siteurl', 'home');

# Reset admin password:
UPDATE wp_users SET user_pass = MD5('newpassword') WHERE user_login = 'admin';

# Find WordPress version:
SELECT option_value FROM wp_options WHERE option_name = 'db_version';

# List all posts:
SELECT ID, post_title, post_status FROM wp_posts WHERE post_type = 'post';

# ============================================================================
# SERVICE MANAGEMENT
# ============================================================================
systemctl status mariadb  # Check MariaDB status
systemctl start mariadb  # Start MariaDB
systemctl stop mariadb  # Stop MariaDB
systemctl restart mariadb  # Restart MariaDB
systemctl enable mariadb  # Enable at boot
journalctl -u mariadb -f  # Follow MariaDB logs

# ============================================================================
# CONFIGURATION
# ============================================================================
# Config file locations:
# - /etc/mysql/my.cnf (Debian/Ubuntu)
# - /etc/my.cnf (RHEL/CentOS)
# - /etc/mysql/mariadb.conf.d/ (MariaDB specific)

# Important settings in my.cnf:
# [mysqld]
# max_connections = 200
# innodb_buffer_pool_size = 1G
# slow_query_log = 1
# slow_query_log_file = /var/log/mysql/slow.log
# long_query_time = 2

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
# Can't connect:
systemctl status mariadb  # Check service running
netstat -tulpn | grep 3306  # Check port listening
cat /etc/mysql/my.cnf | grep bind-address  # Check bind address
mysql -u root -p -e "SELECT User, Host FROM mysql.user;"  # Check user hosts

# Check error log:
tail -f /var/log/mysql/error.log

# Check slow queries:
tail -f /var/log/mysql/slow.log

# "Too many connections" error:
SHOW VARIABLES LIKE 'max_connections';  # Check limit
SHOW PROCESSLIST;  # Check current connections
SET GLOBAL max_connections = 200;  # Increase limit

# Check table corruption:
CHECK TABLE users;
REPAIR TABLE users;

# ============================================================================
# COMMON WORKFLOWS
# ============================================================================
# Create WordPress database:
CREATE DATABASE wordpress CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE USER 'wpuser'@'localhost' IDENTIFIED BY 'password';
GRANT ALL PRIVILEGES ON wordpress.* TO 'wpuser'@'localhost';
FLUSH PRIVILEGES;

# Backup before maintenance:
mysqldump -u root -p --single-transaction --quick --lock-tables=false wordpress | gzip > wordpress_backup_$(date +%Y%m%d).sql.gz

# Restore WordPress database:
gunzip < wordpress_backup.sql.gz | mysql -u root -p wordpress

# Reset WordPress admin password:
mysql -u root -p wordpress -e "UPDATE wp_users SET user_pass = MD5('newpassword') WHERE user_login = 'admin';"

# ============================================================================
# SYNTAX REFERENCE
# ============================================================================
# mysql -u <user> -p [database]
# SHOW <DATABASES|TABLES|PROCESSLIST|STATUS|VARIABLES>
# CREATE <DATABASE|TABLE|USER> ...
# GRANT <privileges> ON <database>.<table> TO 'user'@'host'
# mysqldump -u <user> -p <database> > backup.sql
# mysql -u <user> -p <database> < backup.sql
