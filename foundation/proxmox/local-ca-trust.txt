# -----------------------------------------------
# 6. Trust Proxmox Self-Signed CA on macOS
# -----------------------------------------------
# Proxmox generates a self-signed root CA (pve-root-ca.pem) that signs
# the server certificate (pve-ssl.pem). Trusting the root CA avoids
# browser warnings and covers future certificate renewals automatically.

# -----------------------------------------------
# ISSUE FACED: Certificate has wrong IP address
# -----------------------------------------------
# After changing management IP from 192.168.0.x to 10.0.5.x, the Proxmox
# certificate still showed the old IP in Subject Alternative Name (SAN).
# Browser trusted the hostname but showed warning for IP access.
#
# Check current certificate SAN:
# openssl x509 -in /etc/pve/local/pve-ssl.pem -text | grep -A1 "Subject Alternative Name"
#
# Example (wrong - shows old IP):
# IP Address:127.0.0.1, IP Address:192.168.0.58, DNS:pve-dev, DNS:pve-dev.lab.local
#
# ROOT CAUSE:
# Proxmox reads the node IP from /etc/hosts when generating certificates.
# If /etc/hosts has old IP, certificate will have old IP.
#
# SOLUTION:
# 1. Update /etc/hosts with new management IP
# 2. Regenerate Proxmox certificate
# 3. Restart pveproxy

# -----------------------------------------------
# Fix certificate on DEV server (10.0.5.110)
# -----------------------------------------------
# Step 1: Update /etc/hosts
sed -i 's/192.168.0.58/10.0.5.110/' /etc/hosts
# Or manually: sed -i '/\spve-dev/d' /etc/hosts && echo "10.0.5.110  pve-dev.lab.local pve-dev" >> /etc/hosts

# Step 2: Verify /etc/hosts
cat /etc/hosts

# Step 3: Regenerate certificate
pvecm updatecerts --force

# Step 4: Restart proxy
systemctl restart pveproxy

# Step 5: Verify new certificate has correct IP
openssl x509 -in /etc/pve/local/pve-ssl.pem -text | grep -A1 "Subject Alternative Name"
# Should show: IP Address:10.0.5.110, DNS:pve-dev, DNS:pve-dev.lab.local

# -----------------------------------------------
# Fix certificate on PROD server (10.0.5.100)
# -----------------------------------------------
sed -i 's/192.168.0.[0-9]*/10.0.5.100/' /etc/hosts
pvecm updatecerts --force
systemctl restart pveproxy
openssl x509 -in /etc/pve/local/pve-ssl.pem -text | grep -A1 "Subject Alternative Name"

# -----------------------------------------------
# Trust CA on macOS (after certificate is fixed)
# -----------------------------------------------

# 6a. Copy the root CA from DEV server
scp root@pve-dev.lab.local:/etc/pve/pve-root-ca.pem ~/Desktop/pve-root-ca-dev.pem

# 6b. Copy the root CA from PROD server
scp root@pve-prod.lab.local:/etc/pve/pve-root-ca.pem ~/Desktop/pve-root-ca-prod.pem

# 6c. Add to macOS System Keychain as trusted root
sudo security add-trusted-cert -d -r trustRoot -k /Library/Keychains/System.keychain ~/Desktop/pve-root-ca-dev.pem
sudo security add-trusted-cert -d -r trustRoot -k /Library/Keychains/System.keychain ~/Desktop/pve-root-ca-prod.pem

# 6d. Restart Chrome (fully quit and reopen) for the change to take effect

# Verify the certificate is trusted:
security find-certificate -c "Proxmox Virtual Environment" /Library/Keychains/System.keychain
