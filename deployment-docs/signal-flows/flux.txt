PHASE 1 — GIT PUSH TRIGGERS THE CHAIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[You]
  │
  │  git push origin dev
  │  (any change — new app, updated config, patch file, sync file)
  ▼
[GitHub]
  │
  │  stores new commit on dev branch
  │  assigns new commit SHA
  │  no webhook to Flux — Flux polls, GitHub does not push to Flux
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 2 — SOURCE-CONTROLLER DETECTS AND PULLS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[source-controller]  ← one of four Flux controllers, runs inside k8s
  │
  │  watches GitRepository objects in cluster
  │  your GitRepository object defines:
  │    url: ssh://github.com/mohamedsabry-dev/hybrid-cloud-infrastructure
  │    branch: dev
  │    interval: 1m0s
  │    secretRef: flux-system (holds SSH deploy key)
  │
  │  every 1 minute:
  │    connects to GitHub via SSH using deploy key secret
  │    compares latest remote commit SHA against last known SHA
  │    if same SHA → nothing changed → stops here, waits next interval
  │    if new SHA → change detected → proceeds
  │
  │  pulls new commit from GitHub
  │  downloads full repository content
  │  verifies integrity
  │  creates an Artifact object in cluster:
  │    Artifact = internal Flux object representing a downloaded snapshot
  │    contains: revision (commit SHA), checksum, local storage path
  │    stored inside source-controller pod's local volume
  │
  │  updates GitRepository status:
  │    artifact.revision = new commit SHA
  │    ready = true
  │
  │  notifies watchers that GitRepository has a new Artifact
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 3 — KUSTOMIZE-CONTROLLER RECONCILES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[kustomize-controller]  ← second Flux controller
  │
  │  watches Kustomization objects in cluster
  │  your setup has three Kustomization objects:
  │    flux-system   → path: ./kubernetes/dev/flux
  │    infrastructure → path: ./kubernetes/dev/deployments/infrastructure
  │    apps          → path: ./kubernetes/dev/deployments/apps
  │
  │  sees GitRepository Artifact updated
  │  checks each Kustomization against the new Artifact
  │
  │  DEPENDENCY CHECK (dependsOn):
  │    apps Kustomization has dependsOn: infrastructure
  │    kustomize-controller checks: is infrastructure READY?
  │    if infrastructure not READY → apps reconciliation blocked → waits
  │    if infrastructure READY → apps allowed to proceed
  │
  │  for each Kustomization (in dependency order):
  │
  │    STEP 1 — fetch Artifact
  │      downloads repository snapshot from source-controller local storage
  │      extracts content for the configured path only
  │      e.g. for infrastructure: only reads ./kubernetes/dev/deployments/infrastructure/
  │
  │    STEP 2 — run kustomize build
  │      reads kustomization.yaml (kustomize.config.k8s.io/v1beta1) in that path
  │      resolves all resources listed
  │      applies any patches defined:
  │        patch files read → fields merged on top of base manifests
  │        e.g. flux-pod-anti-affinity.yaml merged onto gotk-components.yaml
  │      produces final list of plain Kubernetes manifest objects
  │
  │    STEP 3 — calculate diff
  │      compares produced manifests against what is currently in cluster
  │      determines: what needs to be created, updated, or deleted
  │
  │    STEP 4 — apply to cluster
  │      sends each manifest to Kubernetes API server
  │      API server validates and writes to etcd
  │      resources are stamped with ownership labels:
  │        kustomize.toolkit.fluxcd.io/name: infrastructure
  │        kustomize.toolkit.fluxcd.io/namespace: flux-system
  │      these labels track which Kustomization owns which resource
  │
  │    STEP 5 — prune check (if prune: true)
  │      compares resources currently stamped with this Kustomization's name
  │      against resources that were just applied
  │      any resource stamped with this name but NOT in latest apply → deleted
  │      this is how removing a file from Git removes the resource from cluster
  │      WARNING: if Kustomization is renamed, old name disappears from Git
  │               prune sees ALL resources stamped with old name as orphans
  │               deletes everything → mass deletion cascade
  │
  │    STEP 6 — health check (if healthChecks defined)
  │      watches specified resources until they are healthy
  │      e.g. healthCheck on vault-agent-injector Deployment:
  │        waits until Deployment has minimum ready replicas
  │        if timeout reached before healthy → Kustomization marked not READY
  │        dependsOn chain blocked → apps cannot proceed
  │      if no healthChecks defined → marks READY immediately after apply
  │
  │    STEP 7 — update Kustomization status
  │      READY: true  → dependsOn unblocked for children
  │      READY: false → dependsOn children remain blocked
  │      revision updated to match current commit SHA
  │
  │  repeats this entire cycle every interval (default 5m0s per Kustomization)
  │  even if no git change → re-applies → drift correction
  │  if someone manually changes something in cluster → Flux reverts it
  │

PHASE 4 — HELM-CONTROLLER HANDLES HELMRELEASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[helm-controller]  ← third Flux controller
  │
  │  watches HelmRelease objects in cluster
  │  your HelmReleases include: vault, ingress-nginx, csi-driver-nfs
  │
  │  HelmRelease object references a HelmRepository:
  │    HelmRepository = source of Helm charts (e.g. helm.releases.hashicorp.com)
  │    source-controller also watches HelmRepository objects
  │    polls chart repository index on interval (e.g. 1h for hashicorp)
  │    downloads and stores chart as Artifact (same Artifact mechanism as Git)
  │
  │  when HelmRelease or chart Artifact changes:
  │
  │    STEP 1 — fetch chart
  │      downloads Helm chart from HelmRepository Artifact
  │      chart version locked to what you specified (e.g. vault 0.32.0)
  │
  │    STEP 2 — render chart with values
  │      runs helm template equivalent internally
  │      merges your values: block from HelmRelease YAML:
  │        global.externalVaultAddr: https://vault.lab.local:8200
  │        server.enabled: false
  │        injector.enabled: true
  │        injector.replicas: 2
  │        injector.nodeSelector: control-plane
  │        injector.affinity: podAntiAffinity across masters
  │      produces final rendered Kubernetes manifests
  │
  │    STEP 3 — install or upgrade
  │      if release does not exist → helm install
  │      if release exists and values/chart changed → helm upgrade
  │      if release exists and nothing changed → no-op
  │      all rendered resources owned by this HelmRelease
  │
  │    STEP 4 — track release
  │      stores helm release secret in same namespace
  │      helm release secret = record of what was installed and at which revision
  │      used for rollback if upgrade fails
  │
  │    CRITICAL: if HelmRelease object is deleted (e.g. by prune):
  │      helm-controller runs helm uninstall
  │      ALL resources created by that helm release are removed
  │      pods, services, configmaps, secrets — everything gone
  │      this is what caused the complete outage in TS-K8S-019
  │
  │  repeats on interval (default 5m0s)
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 5 — NOTIFICATION-CONTROLLER (OPTIONAL BUT PRESENT)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  │

PHASE 6 — OWNERSHIP MODEL (HOW FLUX TRACKS WHAT IT OWNS)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[Every resource Flux applies gets stamped with labels]
  │
  │  kustomize.toolkit.fluxcd.io/name: infrastructure
  │  kustomize.toolkit.fluxcd.io/namespace: flux-system
  │
  │  these labels are stored on the resource in etcd
  │  Flux reads these labels during prune to know what it owns
  │
  │  OWNERSHIP RULES:
  │    one resource can only be owned by one Kustomization at a time
  │    if Kustomization A applies a resource → stamped with A
  │    if you move resource to Kustomization B → B applies it → re-stamped with B
  │    A no longer sees it in its inventory → prune deletes it if prune:true
  │    B now owns it
  │
  │  SAFE MIGRATION REQUIRES:
  │    prune: false on the source Kustomization before any rename or removal
  │    otherwise prune fires on the old name disappearing → mass deletion
  │
  │  INVENTORY:
  │    each Kustomization object stores status.inventory
  │    complete list of every resource it currently owns
  │    used for drift detection and prune calculation
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 7 — DRIFT CORRECTION (CONTINUOUS RECONCILIATION)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[Every interval — even with no git changes]
  │
  │  kustomize-controller re-runs the full reconciliation cycle
  │  compares desired state (Git) against actual state (cluster)
  │
  │  scenarios:
  │
  │  someone runs kubectl edit on a resource manually:
  │    next reconcile → Flux detects drift
  │    re-applies Git version → manual change overwritten
  │    Git is always the source of truth
  │
  │  someone kubectl deletes a resource:
  │    next reconcile → Flux sees resource missing
  │    re-creates it from Git
  │
  │  someone adds a resource manually with kubectl apply:
  │    Flux does not delete it (no ownership label → not in Flux inventory)
  │    it just stays as an unmanaged resource
  │    only prune removes resources → only if they have Flux ownership label
  │
  │  this is the core GitOps guarantee:
  │    cluster always converges back to what Git says
  │    no manual change survives the next reconcile interval
  │

PHASE 8 — FLUX SELF-MANAGEMENT (HOW FLUX MANAGES ITSELF)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[gotk-sync.yaml — bootstrapped by flux bootstrap command]
  │
  │  contains two objects:
  │    GitRepository: flux-system
  │      url: your GitHub repo
  │      branch: dev
  │      interval: 1m
  │      secret: SSH deploy key
  │
  │    Kustomization: flux-system
  │      path: ./kubernetes/dev/flux
  │      interval: 10m
  │      sourceRef: GitRepository flux-system
  │      prune: true
  │
  │  this is the root — everything starts here
  │  flux-system Kustomization watches the flux folder
  │  flux folder contains:
  │    flux-system/ (gotk-components.yaml, gotk-sync.yaml)
  │    infrastructure-sync.yaml
  │    apps-sync.yaml
  │
  │  flux-system Kustomization applies these files
  │  which creates the infrastructure and apps Kustomization objects in cluster
  │  which triggers kustomize-controller to start reconciling those paths
  │
  │  PATCHES ON FLUX CONTROLLERS:
  │    gotk-components.yaml has default Flux controller Deployments
  │    your kustomization.yaml in flux-system/ has patches:
  │      flux-pod-anti-affinity.yaml merged on top
  │      adds replicas: 2 and podAntiAffinity to all Flux controllers
  │    kustomize-controller applies these patches to itself
  │    Flux literally manages and configures its own controllers via GitOps
  │
  │  FLUX UPGRADE:
  │    run flux bootstrap again
  │    gotk-components.yaml regenerated with new controller versions
  │    pushed to Git
  │    flux-system Kustomization detects change
  │    applies new gotk-components.yaml
  │    your patches automatically re-applied on top
  │    Flux upgrades itself with zero manual kubectl commands
  │

THE TWO KUSTOMIZATION TYPES — NEVER CONFUSE THESE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[kustomize.config.k8s.io/v1beta1 — Kustomize Kustomization]
  │  lives in:   kustomization.yaml files inside your repo folders
  │  purpose:    bundle resources and patches together for kustomize build
  │  has:        resources, patches, commonLabels, images
  │  does NOT have: interval, dependsOn, healthChecks, prune, sourceRef
  │  processed by: kustomize-controller internally during build step
  │  never creates a cluster object — only used during build

[kustomize.toolkit.fluxcd.io/v1 — Flux Kustomization]
  │  lives in:   your sync files (infrastructure-sync.yaml, apps-sync.yaml)
  │  purpose:    tell Flux what path to watch, when, with what ordering
  │  has:        interval, dependsOn, healthChecks, prune, sourceRef, path
  │  creates:    an actual object in cluster that Flux watches and reconciles
  │  managed by: kustomize-controller

FAILURE SCENARIOS — WHAT BREAKS AND WHY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Kustomization renamed with prune: true active:
  → old name disappears from Git
  → prune sees all resources stamped with old name as orphans
  → deletes everything → HelmRelease deleted → helm uninstall runs
  → all pods removed → complete outage (TS-K8S-019)
  → fix: set prune: false before any rename or removal

healthCheck references a deleted resource:
  → Kustomization waits forever for resource that no longer exists
  → never becomes READY → dependsOn children blocked
  → complete deadlock → nothing deploys
  → fix: flux suspend kustomization → manually bootstrap → resume

source-controller cannot reach GitHub:
  → no new Artifacts produced
  → kustomize-controller keeps reconciling last known Artifact
  → cluster stays at last successful revision
  → no new changes applied until connectivity restored

kustomize-controller crashes or is restarted:
  → reconciliation pauses until pod recovers
  → since Flux runs with 2 replicas and anti-affinity → other replica takes over
  → no gap in reconciliation if HA configured correctly

Git repo has invalid YAML pushed:
  → kustomize build fails
  → Kustomization marked not READY with error message
  → last successful state remains in cluster (no rollback, just no update)
  → fix the YAML and push → next reconcile succeeds

HelmRelease chart version does not exist in repository:
  → helm-controller cannot pull chart
  → HelmRelease marked not READY
  → existing installed release unchanged (no uninstall on fetch failure)
  → fix: correct the version in HelmRelease YAML

Node running Flux controllers goes down:
  → with single replica → reconciliation stops until pod rescheduled
  → with 2 replicas + anti-affinity across nodes → other replica continues
  → this is why anti-affinity patch on gotk-components matters (TS-K8S-019 lesson)
