================================================================================
VAULT + KUBERNETES — COMPLETE SIGNAL FLOW
From: git push → To: secret injected and live inside pod
================================================================================

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 1 — GIT PUSH & FLUX DETECTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[You]
  │
  │  git push origin dev
  │  (deployment yaml with vault annotations committed)
  ▼
[GitHub]
  │
  │  stores new commit on dev branch
  │
  ▼
[Flux source-controller]  ← runs inside k8s cluster
  │
  │  polls GitHub every 1 minute (GitRepository interval)
  │  detects new commit sha
  │  pulls repo via SSH (flux-system secret holds deploy key)
  │  stores downloaded content as Artifact object in cluster
  │
  ▼
[Flux kustomize-controller]
  │
  │  watches for new Artifact from source-controller
  │  reads kustomization.yaml → resolves which files to apply
  │  runs kustomize build → produces final merged manifests
  │  checks dependsOn → waits for infrastructure READY if configured
  │  sends final Deployment manifest to Kubernetes API server
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 2 — KUBERNETES CONTROL PLANE PROCESSES DEPLOYMENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[Kubernetes API server]
  │
  │  single entry point — nothing touches etcd directly except API server
  │  receives Deployment object from Flux kustomize-controller
  │  validates schema, RBAC permissions, admission controllers
  │  writes Deployment to etcd
  │
  ├──writes──► [etcd]
  │              stores Deployment object as cluster state
  │              notifies watchers that new Deployment exists
  │
  ▼
[Deployment controller]  ← runs inside kube-controller-manager
  │
  │  watches etcd via API server for Deployment objects
  │  sees new Deployment: wordpress, replicas: 3
  │  creates ReplicaSet object
  │  sends ReplicaSet to API server → written to etcd
  │
  ▼
[ReplicaSet controller]
  │
  │  watches for ReplicaSet objects
  │  sees ReplicaSet needs 3 pods, 0 exist
  │  creates 3 Pod spec objects
  │  sends each Pod spec to API server
  │
  ▼
[Kubernetes API server]  ← Pod spec arrives here BEFORE scheduling
  │
  │  receives Pod spec
  │  runs admission controllers in order:
  │    1. mutating admission webhooks   ← THIS IS WHERE VAULT INTERCEPTS
  │    2. validating admission webhooks
  │    3. writes to etcd if all pass
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 3 — MUTATING WEBHOOK INTERCEPT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[API server — MutatingWebhookConfiguration check]
  │
  │  checks: is there a MutatingWebhook registered for Pod resources?
  │  finds: vault-agent-injector registered its webhook on install
  │  (MutatingWebhookConfiguration object exists in cluster from HelmRelease)
  │
  │  forwards FULL pod spec to vault-agent-injector via HTTPS
  │  (injector runs as a Service inside k8s, API server calls it internally)
  │
  ▼
[vault-agent-injector]  ← running on master nodes (2 replicas, anti-affinity)
  │
  │  receives full pod spec from API server
  │  reads pod annotations:
  │
  │    vault.hashicorp.com/agent-inject: "true"         → yes, inject
  │    vault.hashicorp.com/role: "wordpress"            → which vault role
  │    vault.hashicorp.com/agent-inject-secret-db-creds: "secret/data/wordpress/config"
  │                                                     → which secret path
  │                                                     → filename = db-creds
  │    vault.hashicorp.com/agent-inject-template-db-creds: |
  │        {{ with secret "..." }} ... {{ end }}        → how to format output file
  │    vault.hashicorp.com/tls-skip-verify: "false"     → enforce TLS verification
  │    vault.hashicorp.com/tls-secret: "vault-ca"       → mount this K8s Secret
  │    vault.hashicorp.com/ca-cert: "/vault/tls/ca.crt" → use this file for TLS trust
  │
  │  MUTATES the pod spec — adds:
  │    ├── init container: vault-agent-init
  │    │     image: hashicorp/vault-agent
  │    │     config: generated vault-agent config (which secret, which template)
  │    │
  │    ├── sidecar container: vault-agent
  │    │     image: hashicorp/vault-agent
  │    │     same config, runs continuously
  │    │
  │    ├── emptyDir volume: /vault/secrets/
  │    │     shared between vault-agent-init, vault-agent, and main container
  │    │     this is where secret files will be written
  │    │
  │    ├── volume mount: /vault/tls/
  │    │     mounts vault-ca K8s Secret → ca.crt available at /vault/tls/ca.crt
  │    │     used by vault-agent containers to verify vault TLS cert
  │    │
  │    └── vault config volume
  │          generated config file for vault-agent binary
  │
  │  returns mutated pod spec back to API server
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 4 — SCHEDULING AND NODE ASSIGNMENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[API server]
  │
  │  receives mutated pod spec from injector
  │  runs validating webhooks (if any)
  │  writes final mutated pod spec to etcd
  │
  ├──writes──► [etcd]
  │              stores full mutated pod spec
  │              pod status = Pending (no node assigned yet)
  │
  ▼
[kube-scheduler]
  │
  │  watches for Pending pods with no node assigned
  │  evaluates all nodes against:
  │    - resource requests (cpu/memory)
  │    - nodeSelector (e.g. control-plane for vault-agent-injector)
  │    - affinity / anti-affinity rules
  │    - taints and tolerations
  │    - pod topology spread constraints
  │  selects best node (e.g. k8s-worker1)
  │  writes node assignment back to pod spec in etcd via API server
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 5 — KUBELET STARTS THE POD
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[kubelet on k8s-worker1]
  │
  │  agent running on every node (masters and workers)
  │  watches API server for pods assigned to its node
  │  sees pod assigned to k8s-worker1
  │  reads full mutated pod spec from API server
  │
  │  calls containerd (container runtime):
  │    - pull vault-agent-init image if not cached
  │    - pull vault-agent image if not cached
  │    - pull wordpress image if not cached
  │    - create pod sandbox (network namespace, shared volumes)
  │    - mount emptyDir at /vault/secrets/ (shared volume created here)
  │    - mount vault-ca Secret at /vault/tls/ca.crt
  │    - mount pod ServiceAccount token at /var/run/secrets/kubernetes.io/serviceaccount/token
  │      (K8s auto-mounts this into EVERY container — no explicit config needed)
  │
  │  Kubernetes init container rule:
  │    start init containers first, in order
  │    each must exit 0 before next starts
  │    main containers do NOT start until ALL init containers exit 0
  │
  ▼
[containerd]
  │
  │  starts vault-agent-init container
  │


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PHASE 6 — VAULT-AGENT-INIT (init container)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[vault-agent-init]
  │
  │  PURPOSE: guarantee secret file exists BEFORE main container starts
  │
  │  STEP 1 — read pod identity
  │    reads /var/run/secrets/kubernetes.io/serviceaccount/token
  │    this is the pod's own ServiceAccount JWT token
  │    auto-mounted by kubelet, no explicit config needed
  │    identifies this pod as: namespace=apps, serviceaccount=wordpress-sa
  │
  │  STEP 2 — authenticate to Vault (HTTPS, TLS verified via /vault/tls/ca.crt)
  │
  │    POST https://vault.lab.local:8200/v1/auth/kubernetes/login
  │    body: {
  │      jwt: <pod SA token>,
  │      role: "wordpress"
  │    }
  │    TLS: validates vault.lab.local cert against FreeIPA CA at /vault/tls/ca.crt
  │         (no skip-verify — real certificate trust enforced)
  │
  ▼
[Vault cluster — vault.lab.local:8200 → keepalived VIP → active vault node]
  │
  │  receives login request
  │  sees auth method: kubernetes
  │
  │  STEP 3 — call K8s TokenReview API to validate the JWT
  │
  │    POST https://k8s-api:6443/apis/authentication.k8s.io/v1/tokenreviews
  │    Authorization: Bearer <vault-auth-token>
  │    body: { token: <pod SA token> }
  │
  │    vault-auth-token is the long-lived token from:
  │      Secret: vault-auth-token (kube-system)
  │      ServiceAccount: vault-auth (kube-system)
  │      ClusterRoleBinding: system:auth-delegator
  │      → grants vault permission to call TokenReview API
  │
  ▼
[K8s API server — TokenReview]
  │
  │  validates the pod SA token
  │  confirms: token is real, not expired
  │  responds to Vault with:
  │    authenticated: true
  │    username: system:serviceaccount:apps:wordpress-sa
  │    namespace: apps
  │    serviceaccount: wordpress-sa
  │
  ▼
[Vault cluster — continues login processing]
  │
  │  STEP 4 — check Kubernetes role
  │    looks up role named "wordpress" in auth/kubernetes/
  │    role defines:
  │      bound_service_account_names: ["wordpress-sa"]
  │      bound_service_account_namespaces: ["apps"]
  │      policies: ["wordpress-policy"]
  │      token_ttl: 1h (short-lived)
  │
  │  compares K8s TokenReview response against role:
  │    namespace=apps ✓ matches bound namespace
  │    serviceaccount=wordpress-sa ✓ matches bound SA
  │
  │  STEP 5 — issue short-lived Vault token
  │    generates Vault token with wordpress-policy attached
  │    TTL: 1 hour (short-lived — like AWS STS assume role temp token)
  │    returns token to vault-agent-init
  │
  ▼
[vault-agent-init — continues]
  │
  │  STEP 6 — fetch secret using Vault token
  │
  │    GET https://vault.lab.local:8200/v1/secret/data/wordpress/config
  │    X-Vault-Token: <short-lived vault token>
  │
  │    wordpress-policy allows:
  │      path "secret/data/wordpress/config" { capabilities = ["read"] }
  │
  │    Vault returns KV-v2 response:
  │      data.data.db_password = "Change_Me"
  │      data.data.db_user = "wordpress"
  │      metadata.version = 3
  │      metadata.created_time = ...
  │
  │  STEP 7 — render template
  │    applies Go template from annotation:
  │      {{ with secret "secret/data/wordpress/config" }}
  │      WORDPRESS_DB_PASSWORD={{ index .Data.data "db_password" }}
  │      {{ end }}
  │
  │    NOTE: index function required for hyphenated key names
  │    NOTE: without template → raw KV-v2 JSON dump including metadata
  │
  │  STEP 8 — write to shared volume
  │    writes rendered output to /vault/secrets/db-creds
  │    /vault/secrets/ is emptyDir shared with main container
  │    file now exists and is readable by wordpress container
  │
  │  STEP 9 — exits with code 0
  │
  ▼
[kubelet]
  │
  │  sees vault-agent-init exited 0
  │  proceeds to start main containers simultaneously:
  │    ├── wordpress container  (main app)
  │    └── vault-agent sidecar (starts own auth flow in parallel)
  │

PHASE 7 — MAIN CONTAINER STARTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[wordpress container]
  │
  │  starts and reads /vault/secrets/db-creds

PHASE 8 — VAULT-AGENT SIDECAR (runs forever)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ▼
[vault-agent sidecar]
  │
  │  PURPOSE: keep secret file fresh, no pod restart needed for secret changes
  │
  │  runs its OWN independent auth flow (same steps as init):
  │    reads pod SA token → POST to vault login → TokenReview → role check
  │    gets its OWN short-lived Vault token (separate from init's token)
  │
  │  watches secret/data/wordpress/config in Vault
  │  default check interval: ~5 minutes (lease renewal driven)
  │
  │  TOKEN RENEWAL LOOP:
  │    tracks Vault token TTL
  │    before TTL expires → calls vault token renew
  │    if renewal fails → re-authenticates from scratch (full login flow again)
  │    if network lost before renewal:
  │      retries with backoff when connection restored
  │      re-authenticates using pod SA token (always available)
  │      gets new Vault token → resumes watching
  │
  │  ON SECRET CHANGE DETECTED IN VAULT:
  │    fetches new secret value using current Vault token
  │    re-renders template with new values
  │    overwrites /vault/secrets/db-creds with new content
  │    NO pod restart needed
  │    wordpress container reads updated file on next file read
  │
  │  IMPORTANT — put vs patch in Vault:
  │    vault kv put   → replaces ALL keys in secret (missing keys = empty)
  │    vault kv patch → updates only specified key, keeps others intact
  │    if you put with only db_password and template expects db_user too
  │    → db_user renders as empty in file → app breaks
  │    always use patch for partial updates
  │
  │  runs indefinitely until pod is deleted
  │

SUPPORTING INFRASTRUCTURE — TRUST SETUP (one-time, done before any of above)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

[Ansible playbook — integration-vault-k8s-trust.yml]
  │
  │  runs on k8s-master1:
  │    kubectl config view → gets k8s API endpoint
  │    kubectl config view --raw → gets k8s CA certificate
  │    kubectl create token vault-auth --duration=30000h → long-lived SA token
  │    kubectl get /.well-known/openid-configuration → gets OIDC issuer URL
  │
  │  runs on vault1:
  │    vault auth enable kubernetes
  │    vault write auth/kubernetes/config
  │      kubernetes_host = https://k8s-api:6443
  │      kubernetes_ca_cert = k8s CA cert
  │      token_reviewer_jwt = vault-auth long-lived token
  │      issuer = k8s OIDC issuer URL
  │
  │  result:
  │    Vault now trusts K8s tokens
  │    Vault can call K8s TokenReview API using vault-auth token
  │    vault-auth has system:auth-delegator → allowed to validate any pod token

[Per-app setup — vault-pod-setup.sh on vault1]
  │
  │  prompts for: app name, namespace, SA name, secret key/values
  │  creates in Vault:
  │    secret/<app>/config → KV-v2 secret with provided values
  │    <app>-policy → read-only policy on that secret path
  │    Kubernetes role <app>:
  │      bound_service_account_names: [<sa-name>]
  │      bound_service_account_namespaces: [<namespace>]
  │      policies: [<app>-policy]
  │
  │  result: one role per app, scoped to exact namespace + SA
  │  wordpress-sa in apps namespace can only read wordpress secrets
  │  no other pod can use that role

[vault-auth K8s objects — in cluster]
  │
  │  ServiceAccount: vault-auth (kube-system)
  │    identity that Vault uses to call K8s
  │
  │  Secret: vault-auth-token (kube-system)
  │    annotation: kubernetes.io/service-account.name: vault-auth
  │    K8s auto-binds a long-lived token to this secret
  │    required because K8s 1.24+ no longer auto-creates SA tokens
  │
  │  ClusterRoleBinding: vault-auth
  │    binds vault-auth SA to system:auth-delegator ClusterRole
  │    system:auth-delegator = permission to call TokenReview API
  │    without this: K8s rejects Vault's token validation calls → auth fails
