================================================================================
                         INSTALLATION STEPS
================================================================================

Step 1: Install Dependencies
─────────────────────────────────────────────────────────────────────────────────
# Update system
dnf update -y

# Install required packages
dnf install -y curl tar git jq docker

# Enable Docker
systemctl enable --now docker

# Install additional tools (if needed)
dnf install -y python3 python3-pip

Step 2: Create Runner User
─────────────────────────────────────────────────────────────────────────────────
# Create dedicated user for runner
useradd -m -s /bin/bash github-runner
usermod -aG docker github-runner

Step 3: Download and Install Runner
─────────────────────────────────────────────────────────────────────────────────
# Switch to runner user
su - github-runner

# Create runner directory
mkdir -p ~/actions-runner && cd ~/actions-runner

# Download latest runner (x64 Linux)
curl -o actions-runner-linux-x64.tar.gz -L \
  https://github.com/actions/runner/releases/download/v2.331.0/actions-runner-linux-x64-2.331.0.tar.gz

# Extract
tar xzf ./actions-runner-linux-x64.tar.gz

Step 4: Configure Runner
─────────────────────────────────────────────────────────────────────────────────
# Get token from: GitHub Repo > Settings > Actions > Runners > New self-hosted runner

# For DEV runner:
./config.sh --url https://github.com/mohamedsabry-dev/hybrid-cloud-infrastructure \
  --token YOUR_RUNNER_TOKEN \
  --labels self-hosted,linux,x64,dev \
  --name gh-runner-dev

# For PROD runner:
./config.sh --url https://github.com/mohamedsabry-dev/hybrid-cloud-infrastructure \
  --token YOUR_RUNNER_TOKEN \
  --labels self-hosted,linux,x64,prod \
  --name gh-runner-prod

Step 5: Install as Service
─────────────────────────────────────────────────────────────────────────────────
# Exit to root
exit

# Install service (as root)
cd /home/github-runner/actions-runner
./svc.sh install github-runner

# Start service
./svc.sh start

# Enable on boot
systemctl enable actions.runner.*
