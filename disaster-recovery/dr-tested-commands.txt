################################################################################
# ETCD BACKUP & RESTORE
################################################################################

# --- Trigger Manual Backup ---

kubectl create job --from=cronjob/etcd-backup etcd-backup-dr-test -n etcd-backup
# Create manual etcd backup job from existing cronjob

kubectl logs -n etcd-backup -l job-name=etcd-backup-dr-test -f
# Watch backup job logs

# --- Download and Validate S3 Snapshot ---

aws s3 cp s3://hybrid-cloud-k8s-etcd-backup-dev/etcd-YYYYMMDD-HHMMSS.snap /tmp/
# Download snapshot from S3

etcdutl snapshot status /tmp/etcd-YYYYMMDD-HHMMSS.snap --write-out=table
# Validate snapshot integrity (etcd 3.6+ uses etcdutl, not etcdctl)

# --- etcd Cluster Health ---

etcdctl member list
# List all etcd cluster members

etcdctl endpoint health --cluster
# Check health of all etcd endpoints

etcdctl endpoint status --cluster --write-out=table
# Detailed status of all endpoints

################################################################################
# ETCD SINGLE NODE FAILURE & RECOVERY
################################################################################

# --- Stop etcd on a Master (kubeadm static pod) ---

sudo mv /etc/kubernetes/manifests/etcd.yaml /tmp/
# Stop etcd by removing static pod manifest (kubeadm clusters)
# NOTE: Do NOT use systemctl - etcd runs as static pod in kubeadm

# --- Verify Quorum with 2/3 Nodes ---

etcdctl endpoint health --cluster
# One node will show unhealthy, cluster still operational with 2/3

kubectl get nodes
# Affected master shows NotReady, others remain Ready

# --- Single Node Recovery (cluster sync, not S3 restore) ---

# Step 1: Delete corrupted etcd data
sudo rm -rf /var/lib/etcd

# Step 2: Remove broken member from cluster (run on healthy master)
etcdctl member remove <MEMBER_ID>

# Step 3: Re-add member to cluster
etcdctl member add k8s-master3.lab.local --peer-urls=https://10.0.61.12:2380

# Step 4: Ensure manifest has correct state
sudo sed -i 's/--initial-cluster-state=new/--initial-cluster-state=existing/' /tmp/etcd.yaml

# Step 5: Restore static pod manifest
sudo mv /tmp/etcd.yaml /etc/kubernetes/manifests/

# Step 6: Restart kubelet (often needed after etcd recovery)
sudo systemctl restart kubelet

# Step 7: Verify recovery
etcdctl member list
etcdctl endpoint health --cluster
kubectl get nodes

################################################################################
# VAULT CLUSTER OPERATIONS
################################################################################

# --- Vault Status ---

vault status
# Check seal status, HA mode, cluster info

source /etc/profile.d/vault.sh && vault status
# Set VAULT_ADDR first if needed

# --- Raft Cluster ---

vault operator raft list-peers
# List configured raft members (NOT live status)

vault operator raft autopilot state
# ACTUAL live health - shows Healthy, Failure Tolerance, Last Contact

# --- Quorum Loss Test ---

pct stop 2004 && pct stop 2005
# Stop 2 vault LXCs (leader + follower) to test quorum loss

pct start 2004
# Restore quorum (start 1 node = 2/3 = majority)

pct start 2005
# Full restore (3/3 nodes)

# --- Check Vault from Surviving Node ---

vault operator raft autopilot state
# Shows: Healthy: false, Failure Tolerance: 0 when degraded

################################################################################
# VAULT AWS KMS AUTO-UNSEAL
################################################################################

# --- Check Seal Configuration ---

cat /etc/vault.d/vault.hcl | grep -A10 'seal'
# View seal stanza (awskms, region, key_id)

systemctl show vault | grep -i environment
# Check environment file path

cat /etc/vault.d/vault.env
# View AWS credentials location

# --- Break/Restore AWS Credentials (for testing) ---

cp /etc/vault.d/vault.env /etc/vault.d/vault.env.backup
# Backup credentials before testing

echo "# BROKEN FOR DR TEST" > /etc/vault.d/vault.env
# Break credentials (vault will fail to start)

mv /etc/vault.d/vault.env.backup /etc/vault.d/vault.env
# Restore credentials

systemctl restart vault
# Restart after restoring - auto-unseal should work

# --- Check Recovery Logs ---

journalctl -u vault --since "5 minutes ago" | grep -i "unseal\|error\|credential"
# Check for auto-unseal success or credential errors

################################################################################
# KUBERNETES POD KILL TESTS
################################################################################

# --- Kill Single Pod ---

kubectl delete pod -n apps <pod-name>
# Delete specific pod (deployment recreates it)

kubectl delete pod -n apps -l app=wordpress
# Delete all pods matching label

kubectl get pods -n apps -w
# Watch pod recovery in real-time

# --- Scale Deployment ---

kubectl scale deployment wordpress -n apps --replicas=0
# Scale to zero (kill all pods)

kubectl scale deployment wordpress -n apps --replicas=3
# Scale back up

# --- Pod Recovery Timeline ---

kubectl get pods -n apps -o wide -w
# Watch: Init:0/2 -> Init:1/2 -> PodInitializing -> 1/2 -> 2/2

# --- Check Vault Agent Injection ---

kubectl logs <pod-name> -n apps -c vault-agent-init
# Check vault-agent init container logs

kubectl logs <pod-name> -n apps -c vault-agent
# Check vault-agent sidecar logs

# --- Simultaneous Kill (Race Condition Test) ---

kubectl delete pod -n vault -l app.kubernetes.io/name=vault-agent-injector & \
kubectl delete pod -n apps -l app=wordpress
# Kill injector AND app pods simultaneously
# With 2 injector replicas: pods get 2/2 (sidecar injected)
# With 1 injector replica: pods get 1/1 (NO sidecar - DB auth fails)

USE wordpress;
SELECT ID, post_title, post_status, post_date
FROM wp_posts
WHERE post_type='attachment'
ORDER BY post_date DESC
LIMIT 10;

# --- Check for Orphaned/Corrupted Records ---

SELECT ID, post_title, post_status
FROM wp_posts
WHERE post_status NOT IN ('publish','inherit','draft','auto-draft')
ORDER BY ID DESC
LIMIT 10;
# Should return empty set if no corruption

################################################################################
# NFS STORAGE OUTAGE
################################################################################

# --- Check NFS Mounts ---

mount | grep nfs
# List current NFS mounts

showmount -e 10.0.40.120
# List NFS exports from server

df -h /path/to/nfs/mount
# Check mount disk usage

# --- Simulate NFS Outage (Interface Down) ---

ip link set eth1 down
# Take down storage interface on worker

ip link show eth1 | grep state
# Verify: state DOWN

# --- Restore NFS ---

ip link set eth1 up
# Bring interface back up

ip link show eth1 | grep state
# Verify: state UP

showmount -e 10.0.40.120
# Verify NFS server reachable again

# --- Check Kernel NFS Messages ---

dmesg | grep nfs
# Look for "server not responding, timed out" during outage
# Should clear after restore

# --- Check Service Endpoints ---

kubectl get endpoints wordpress -n apps
# During outage: broken pod should be REMOVED from endpoints
# After fix: readiness probe on /wp-content/index.php detects NFS failure
