================================================================================
                         5. CLI CONFIG REFERENCE
================================================================================

   Exported from: network/backups/switchCfg.cfg

   # VLANs
   vlan 40                    # Storage
   vlan 50-55                 # Prod Services
   vlan 60-65                 # Dev Services
   vlan 99                    # Dump/Isolation

   # Management Interface
   interface vlan 1
     ip address 192.168.100.185 255.255.255.0 gateway 192.168.100.1

   # Example Trunk Port (Port2 - Prod)
   interface gigabitEthernet 1/0/2
     switchport general allowed vlan 99 untagged
     switchport general allowed vlan 50-55 tagged
     switchport pvid 99
     no switchport general allowed vlan 1
     power inline supply disable

   # Example Trunk Port (Port3 - Dev)
   interface gigabitEthernet 1/0/3
     switchport general allowed vlan 99 untagged
     switchport general allowed vlan 60-65 tagged
     switchport pvid 99
     no switchport general allowed vlan 1
     power inline supply disable

   # Example Storage Port (Port6)
   interface gigabitEthernet 1/0/6
     switchport general allowed vlan 99 untagged
     switchport general allowed vlan 40 tagged
     switchport pvid 99
     loopback-detection config process-mode port-based recovery-mode auto
     loopback-detection
