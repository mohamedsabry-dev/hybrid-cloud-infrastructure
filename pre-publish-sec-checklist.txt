# Pre-Publish Security Checklist
# Run all checks before making repo public
# Priority: Critical first, then Important, then Good Practice

===============================================================================
CRITICAL — MUST DO BEFORE PUBLISH
===============================================================================

-------------------------------------------------------------------------------
1. GITLEAKS — Scan full git history for secrets
-------------------------------------------------------------------------------
brew install gitleaks

# Scan entire git history — not just current files
# A deleted file 3 months ago still exists in git history
gitleaks detect --source . --log-opts="--all"

# If anything found:
# → DO NOT just delete and push — history still contains it
# → Must rewrite history: git filter-repo
# → AND rotate the exposed credential immediately

-------------------------------------------------------------------------------
2. TERRAFORM STATE FILES — Must never be in repo
-------------------------------------------------------------------------------
# Check for any state files accidentally committed
find . -name "*.tfstate" -o -name "*.tfstate.backup"

# Verify .gitignore covers them
cat .gitignore | grep tfstate

# Expected: no .tfstate files found
# State must live in S3 only — verify this is the case

-------------------------------------------------------------------------------
3. ANSIBLE VAULT FILES — Confirm encrypted not plaintext
-------------------------------------------------------------------------------
# Find any vault files
find . -name "*.vault" -o -name "vault.yml" -o -name "*vault*"

# For each file found — check it is actually encrypted
head -1 <vault-file>
# GOOD:  $ANSIBLE_VAULT;1.1;AES256
# BAD:   any plaintext content = critical problem

-------------------------------------------------------------------------------
4. GITHUB SECRET SCANNING + PUSH PROTECTION — Enable now
-------------------------------------------------------------------------------
# Location: GitHub → Settings → Security → Code security and analysis
#
# Enable ALL of:
#   → Secret scanning
#   → Push protection (blocks pushes containing detected secrets)
#   → Dependabot alerts
#
# Push protection is most important — prevents future accidents

-------------------------------------------------------------------------------
5. GITIGNORE — Verify comprehensive coverage
-------------------------------------------------------------------------------
cat .gitignore

# Must include at minimum:
# *.tfstate
# *.tfstate.backup
# .terraform/
# *.pem
# *.key
# *.p12
# *.pfx
# .env
# .env.*
# ansible/vault_pass*
# *.log
# kubeconfig
# *.kubeconfig

-------------------------------------------------------------------------------
6. WORKFLOW LOGS — Check for exposed private keys or secrets via Terraform
-------------------------------------------------------------------------------
# IMPORTANT: Terraform sometimes prints sensitive values in plan/apply output
# Even with ::add-mask:: — check historical workflow run logs manually
#
# Go to: GitHub → Actions → each workflow → check recent run logs
#
# Look for:
#   → Any private key content (BEGIN PRIVATE KEY, BEGIN RSA PRIVATE KEY)
#   → Any unmasked tokens or passwords
#   → Terraform plan output showing sensitive resource attributes
#   → SSH key content printed in logs
#
# Terraform specific risk:
#   → terraform plan can expose sensitive values if not marked sensitive
#   → check all terraform output blocks: sensitive = true where needed
#   → check all variable definitions: sensitive = true for passwords/tokens
#
# Command to check terraform for missing sensitive flags:
grep -r "output" terraform/ --include="*.tf" -A 5 | grep -v "sensitive"
grep -r 'variable' terraform/ --include="*.tf" -A 5 | grep -i "password\|token\|secret\|key" | grep -v "sensitive"

===============================================================================
IMPORTANT — DO BEFORE PUBLISH
===============================================================================

-------------------------------------------------------------------------------
7. INTERNAL IP REVIEW — Conscious decision required
-------------------------------------------------------------------------------
# Internal IPs are low risk (private ranges, useless without network access)
# But review what is exposed and make a conscious decision

grep -r "10\.0\." . --include="*.yaml" --include="*.yml" --include="*.md" --include="*.tf" | grep -v ".git"
grep -r "192\.168\." . --include="*.yaml" --include="*.yml" --include="*.md" --include="*.tf" | grep -v ".git"

# Decision: internal IPs in config files are acceptable for portfolio
# Decision: internal IPs in documentation are acceptable
# Not acceptable: internal IPs combined with credentials

-------------------------------------------------------------------------------
8. KUBERNETES SECRET MANIFESTS — Check for real values
-------------------------------------------------------------------------------
# Find all Secret kind files
grep -r "kind: Secret" . --include="*.yaml" -l

# For each file found — verify:
#   → stringData contains no real passwords or tokens
#   → data contains no real base64-encoded credentials
#   → vault-ca Secret contains only CA certificate (public by nature — OK)
#   → no MYSQL_ROOT_PASSWORD or similar hardcoded

grep -r "stringData\|data:" . --include="*.yaml" -A 10 | grep -v ".git"

-------------------------------------------------------------------------------
9. SSH KEYS AND PRIVATE KEYS — Scan for accidental commits
-------------------------------------------------------------------------------
# Find any private key files
find . -name "id_rsa" -o -name "id_ed25519" -o -name "*.pem" -o -name "*.key" | grep -v ".git"

# Check for private key headers in any file type
grep -r "BEGIN.*PRIVATE KEY" . --include="*.yaml" --include="*.yml" --include="*.md" --include="*.tf" --include="*.txt" | grep -v ".git"
grep -r "BEGIN RSA PRIVATE KEY" . | grep -v ".git"
grep -r "BEGIN EC PRIVATE KEY" . | grep -v ".git"
grep -r "BEGIN OPENSSH PRIVATE KEY" . | grep -v ".git"

-------------------------------------------------------------------------------
10. COMMIT MESSAGE REVIEW — Check for sensitive content in messages
-------------------------------------------------------------------------------
# Review recent commit messages
git log --oneline -100

# Search all commit messages for sensitive patterns
git log --all --oneline | grep -i "password\|token\|secret\|key\|credential"

# If found in message but not in files — still visible in history
# Rewrite history if needed: git filter-repo --message-callback

-------------------------------------------------------------------------------
11. EMAIL AND PERSONAL INFO — Review what is exposed
-------------------------------------------------------------------------------
# Check for email addresses
grep -r "@gmail\|@hotmail\|@yahoo" . --include="*.yaml" --include="*.yml" --include="*.md" --include="*.tf" | grep -v ".git"

# Check workflow files for hardcoded personal values
grep -r "env:" .github/workflows/ -A 5 | grep -v "secrets\." | grep -v "vars\." | grep -v "github\." | grep -v ".git"

# Every sensitive value must come from secrets.X or vars.X
# Never hardcoded directly in workflow file

===============================================================================
GOOD PRACTICE — CAN DO AFTER PUBLISH
===============================================================================

-------------------------------------------------------------------------------
12. README FRESH EYES REVIEW
-------------------------------------------------------------------------------
# Read all markdown files asking:
#   → Does this reveal anything unintended?
#   → Any customer or employer references that should not be public?
#   → Any internal hostnames revealing sensitive topology?
#   → Any personal details used unexpectedly?

find . -name "*.md" | grep -v ".git" | xargs ls -la

-------------------------------------------------------------------------------
13. DEPENDABOT — Enable for dependency vulnerability alerts
-------------------------------------------------------------------------------
# Location: GitHub → Settings → Security → Code security and analysis
# Enable: Dependabot alerts
# Enable: Dependabot security updates
#
# This monitors your dependencies for known CVEs
# Free and automatic — no reason not to enable

-------------------------------------------------------------------------------
14. DOCKER IMAGE REVIEW — Confirm no sensitive content baked in
-------------------------------------------------------------------------------
# For each Dockerfile in kubernetes/docker-images/
find . -name "Dockerfile" | grep -v ".git"

# For each one verify:
#   → No COPY of credential files
#   → No ENV with passwords or tokens
#   → No RUN commands that fetch credentials
#   → All config comes from ConfigMap or Vault at runtime

grep -r "ENV\|COPY\|ARG" kubernetes/docker-images/ 2>/dev/null

===============================================================================
FINAL VERIFICATION BEFORE CLICKING MAKE PUBLIC
===============================================================================

Run this combined check last:

# 1. No state files
find . -name "*.tfstate" | grep -v ".git"

# 2. No private keys
grep -r "BEGIN.*PRIVATE KEY" . | grep -v ".git"

# 3. No hardcoded secrets in workflows
grep -r "password\|token\|secret" .github/workflows/ | grep -v "secrets\.\|vars\.\|#\|mask" | grep -v ".git"

# 4. Gitleaks clean
gitleaks detect --source . --log-opts="--all"

# If all four return empty or clean → safe to publish
