## Proxmox Pre-Configuration (PVE 9.1)
## Run these commands on the Proxmox host via SSH or console

# -----------------------------------------------
# 1. Create Terraform Role (full VM, CT, Storage, Network privileges)
# -----------------------------------------------
pveum role add TerraformFullRole -privs "VM.Allocate VM.Audit VM.Backup VM.Clone VM.Config.CDROM VM.Config.CPU VM.Config.Cloudinit VM.Config.Disk VM.Config.HWType VM.Config.Memory VM.Config.Network VM.Config.Options VM.Console VM.Migrate VM.PowerMgmt VM.Snapshot VM.Snapshot.Rollback VM.GuestAgent.Audit VM.GuestAgent.FileRead VM.GuestAgent.FileWrite VM.GuestAgent.FileSystemMgmt VM.GuestAgent.Unrestricted Datastore.Allocate Datastore.AllocateSpace Datastore.AllocateTemplate Datastore.Audit SDN.Allocate SDN.Audit SDN.Use Sys.Audit Sys.Modify Sys.Console Pool.Allocate Pool.Audit Mapping.Audit Mapping.Modify Mapping.Use"

# -----------------------------------------------
# 2. Create Terraform User
# -----------------------------------------------
pveum user add terraform@pve -comment "Terraform automation"

# -----------------------------------------------
# 3. Assign Role to User (on root path / = all resources)
# -----------------------------------------------
pveum aclmod / -user terraform@pve -role TerraformFullRole

# -----------------------------------------------
# 4. Create API Token (privsep=0 inherits user permissions)
# -----------------------------------------------
pveum user token add terraform@pve terraform-token --privsep 0

# -----------------------------------------------
# Token Output
# -----------------------------------------------
#  full-tokenid : terraform@pve!terraform-token
#  privsep      : 0
#  secret       : <stored in AWS Secrets Manager: infra/proxmox/api-token>

# -----------------------------------------------
# 5. Test API Token (run from local machine, single line)
# -----------------------------------------------
# Replace <SECRET> with the token secret from AWS Secrets Manager
curl -k -H 'Authorization: PVEAPIToken=terraform@pve!terraform-token=<SECRET>' https://192.168.0.120:8006/api2/json/version

# Expected response: {"data":{"release":"9.1","version":"9.1.5","repoid":"..."}}
